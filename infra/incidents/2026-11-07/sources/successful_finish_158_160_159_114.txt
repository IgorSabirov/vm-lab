fd, temporary = tempfile.mkstemp(prefix='.server-', dir=root)
with os.fdopen(fd, 'w') as output:

    output.flush()

def rpc(method, params=None):
    connection = http.client.HTTPSConnection(host, timeout=8)
    try:
        connection.sock = ssl.create_default_context().wrap_socket(
            socket.create_connection(('127.0.0.1', 443), timeout=8), server_hostname=host)
        connection.request('POST', '/' + cfg['token'] + '/mcp',
            json.dumps({'jsonrpc': '2.0', 'id': 1, 'method': method, 'params': params or {}}),
            {'Content-Type': 'application/json', 'Accept': 'application/json'})
        response = connection.getresponse()
        if response.status != 200:
            raise RuntimeError('MCP HTTP ' + str(response.status))
        return json.loads(response.read())['result']
    finally:
        connection.close()

for attempt in range(15):
    try:
        names = {item['name'] for item in rpc('tools/list')['tools']}
        break
    except OSError:
        if attempt == 14:
            raise
        time.sleep(1)
if not {'write_work_file', 'start_python_job', 'job_status'} <= names:
    raise RuntimeError('Проверка инструментов MCP не пройдена')
status = rpc('tools/call', {'name': 'work_status', 'arguments': {}})
if not json.loads(status['content'][0]['text'])['worker']['responsive']:
    raise RuntimeError('Worker ещё не отвечает')
run('systemctl', 'enable', '--now', 'vm-compute-worker.service',
    'vm-compute-api.service', 'vm-compute-cert-renew.timer')
print('ГОТОВО: HTTPS, MCP и worker проверены локально на ВМ.')
print('URL ДЛЯ ПЛАГИНА: https://' + host + '/' + cfg['token'] + '/mcp')
VM_FINISH
