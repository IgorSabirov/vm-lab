def random_rhs(space,seed):
    rng=np.random.default_rng(seed);rho=np.zeros((space.d,space.d),complex)
    for g in space.groups:
        z=rng.normal(size=(len(g),len(g)))+1j*rng.normal(size=(len(g),len(g)))
        rho[np.ix_(g,g)]=(z+z.conj().T)/2
    rho-=np.trace(rho)*np.eye(space.d)/space.d
    x=space.pack(rho);x/=np.linalg.norm(x)
    return x

def equal(a,b):
    if isinstance(a,dict) and isinstance(b,dict):return a.keys()==b.keys() and all(equal(a[k],b[k]) for k in a)
    if isinstance(a,list) and isinstance(b,list):return len(a)==len(b) and all(equal(x,y) for x,y in zip(a,b))
    if a is None or b is None or isinstance(a,bool) or isinstance(b,bool):return type(a)==type(b) and a==b
    if isinstance(a,(int,float)) and isinstance(b,(int,float)):return bool(np.isfinite([a,b]).all() and np.isclose(a,b,atol=1e-12,rtol=1e-11))
    return a==b


def vector(report,name):

    return x

"""Bordered shifted sparse factors + rank-20 Woodbury; independent GMRES/ILU.

Every factor application is counted, including columns, residual corrections,
preconditioner applications and inverse-norm estimation. No eigenspectrum repair.
"""
import time,warnings
import numpy as np
from scipy import sparse as sp
from scipy import linalg
from scipy.sparse.linalg import splu,spilu,LinearOperator,onenormest
from e002e_exp.compat import gmres
from e002_exp.model import maxabs


class Solver:
    def __init__(self,L,space,gamma=0.,reset=None,method='lu',rtol=1e-11,atol=1e-13,
                 ilu_drop_tol=1e-5,ilu_fill_factor=10.,restart=100,maxiter=400):
        self.L=sp.csr_matrix(L);self.space=space;self.gamma=float(gamma);self.reset=reset
        if not np.isfinite(gamma) or gamma<0 or (gamma and reset is None):raise ValueError('Invalid reset solve')
        if method not in ('lu','gmres_ilu'):raise ValueError('Unknown method')
        self.method=method;self.rtol=rtol;self.atol=atol;self.restart=restart;self.maxiter=maxiter
        self.counts=dict(task_solves=0,factor_calls=0,factor_rhs_columns=0,preconditioner_calls=0,
                         residual_corrections=0,gmres_iterations=0,condition_estimates=0)
        self.times=dict(factor_seconds=0.,low_rank_setup_seconds=0.,factor_application_seconds=0.,solve_seconds=0.,condition_seconds=0.)
        n=space.size;r=space.trace_row.T/space.d
        self.base=sp.bmat([[self.L-gamma*sp.eye(n),r],[space.trace_row,None]],format='csc')
        tick=time.perf_counter()
        self.factor=splu(self.base) if method=='lu' else spilu(self.base,drop_tol=ilu_drop_tol,fill_factor=ilu_fill_factor)
        self.times['factor_seconds']=time.perf_counter()-tick
        self.U=sp.csc_matrix((n+1,0));self.V=sp.csr_matrix((0,n+1));self.W=None;self.small=None
        tick=time.perf_counter()
        if gamma:
            self.U=sp.vstack([reset.U,sp.csr_matrix((1,reset.rank))],format='csc')
            self.V=sp.hstack([reset.V,sp.csr_matrix((reset.rank,1))],format='csr')
            self.W=self.factor_apply(self.U.toarray())
            self.small=np.eye(reset.rank)+gamma*(self.V@self.W)
            if not np.isfinite(self.small).all():raise ArithmeticError('Invalid Woodbury core')
            with warnings.catch_warnings():
                warnings.simplefilter('error',linalg.LinAlgWarning)
                self.small_factor=linalg.lu_factor(self.small)
            if not np.isfinite(self.small_factor[0]).all():raise ArithmeticError('Invalid Woodbury factor')
        self.times['low_rank_setup_seconds']=time.perf_counter()-tick
        self.operator=LinearOperator(self.base.shape,matvec=self.apply,matmat=self.apply,dtype=complex)
        self.preconditioner=LinearOperator(self.base.shape,matvec=self.precondition,dtype=complex)

    def factor_apply(self,rhs,trans='N'):
        tick=time.perf_counter();self.counts['factor_calls']+=1
        self.counts['factor_rhs_columns']+=1 if np.ndim(rhs)==1 else rhs.shape[1]
        out=self.factor.solve(np.asarray(rhs,complex),trans=trans)
        self.times['factor_application_seconds']+=time.perf_counter()-tick
        return out

    def inverse(self,rhs,adjoint=False):
        z=self.factor_apply(rhs,'H' if adjoint else 'N')
        if self.gamma:
            if adjoint:
                c=linalg.lu_solve(self.small_factor,self.U.getH()@z,trans=2)
                z-=self.gamma*self.factor_apply(self.V.getH()@c,'H')
            else:
                c=linalg.lu_solve(self.small_factor,self.V@z)
                z-=self.gamma*(self.W@c)
        if not np.isfinite(z).all():raise ArithmeticError('Nonfinite inverse application')
        return z

    def precondition(self,rhs):
        self.counts['preconditioner_calls']+=1
        return self.inverse(rhs)

    def apply(self,x):
        return self.base@x+self.gamma*(self.U@(self.V@x))

    def equation(self,x):
        return self.L@x+(self.gamma*(self.reset.apply(x)-x) if self.gamma else 0)

    def solve(self,b=None,trace=0.):
        b=np.zeros(self.space.size,complex) if b is None else np.asarray(b,complex)
        if b.shape!=(self.space.size,) or not np.isfinite(b).all():raise ValueError('Invalid RHS')
        if abs(self.space.trace(b))>1e-11:raise ValueError('RHS is not traceless')
        rhs=np.r_[b,trace];tick=time.perf_counter();self.counts['task_solves']+=1
        if self.method=='lu':
            x=self.inverse(rhs)
            for _ in range(2):
                x+=self.inverse(rhs-self.apply(x));self.counts['residual_corrections']+=1
        else:
            def count(_):self.counts['gmres_iterations']+=1
            x,info=gmres(self.operator,rhs,M=self.preconditioner,rtol=self.rtol,atol=self.atol,
                         restart=self.restart,maxiter=self.maxiter,callback=count,callback_type='pr_norm')
            if info!=0:raise ArithmeticError('Reset GMRES did not converge: '+str(info))
        self.times['solve_seconds']+=time.perf_counter()-tick
        residual=self.equation(x[:-1])-b;bnorm=float(np.linalg.norm(b));rnorm=float(np.linalg.norm(residual))
        checks=dict(absolute_inf=maxabs(residual),absolute_2=rnorm,
                    relative_true=None if bnorm==0 else rnorm/bnorm,
                    relative_true_status='ZERO_RHS' if bnorm==0 else 'DEFINED',
                    relative_normalized=rnorm/max(bnorm,1.),trace_error=float(abs(self.space.trace(x[:-1])-trace)),
                    border_multiplier=float(abs(x[-1])))
        if (not np.isfinite(x).all() or max(checks['absolute_inf'],checks['trace_error'],checks['border_multiplier'])>1e-9):
            raise ArithmeticError('Original reset equation/trace failed: '+str(checks))
        return x[:-1],checks

    def condition_estimate(self):
        if self.method!='lu':return dict(value=None,status='NOT_ESTIMATED_FOR_ILU')
        tick=time.perf_counter();self.counts['condition_estimates']+=1
        inverse=LinearOperator(self.base.shape,matvec=self.inverse,
            rmatvec=lambda x:self.inverse(x,adjoint=True),matmat=self.inverse,dtype=complex)
        # Forming rank-20 update for a norm is cheap; it is never factorized.
        matrix=self.base+self.gamma*(self.U@self.V)
        value=float(sp.linalg.norm(matrix,1)*onenormest(inverse))
        self.times['condition_seconds']+=time.perf_counter()-tick
        if not np.isfinite(value):raise ArithmeticError('Nonfinite condition diagnostic')
        return dict(value=value,status='ONE_NORM_ESTIMATE_NOT_RIGOROUS_BOUND')

    def metrics(self):
        factor_bytes=sum(a.data.nbytes+a.indices.nbytes+a.indptr.nbytes for a in (self.factor.L,self.factor.U))
        return dict(method=self.method,gamma=self.gamma,base_nnz=int(self.base.nnz),
                    factor_nnz=int(self.factor.L.nnz+self.factor.U.nnz),factor_storage_bytes=int(factor_bytes),
                    woodbury_rank=0 if self.reset is None or not self.gamma else self.reset.rank,
                    low_rank_dense_bytes=0 if self.W is None else self.W.nbytes,
                    counts=dict(self.counts),timing=dict(self.times),
                    timing_note='Factor-application time is a subcomponent; do not add twice.')

        raise ValueError('VM checkout/source identity mismatch')

            raise ValueError('Duplicate/mismatched evidence')
        records[item['case_id']] = record
    if set(records) != {c['id'] for c in cases()}:
        raise ValueError('Incomplete baseline evidence')
    for case in cases():

            raise ValueError('Baseline physics mismatch')

def memory_preflight(budget, current):
    required = ('peak_additional_bytes', 'reserve_bytes', 'measured_peak_rss_bytes', 'simultaneous_factor_storage_bytes')
    if any(isinstance(budget.get(k), bool) or not isinstance(budget.get(k), (int, float)) or not np.isfinite(budget[k]) or budget[k] <= 0 for k in required):
        raise ValueError('Memory budget lacks measured peak/factor storage and positive reserve')
    if budget['peak_additional_bytes'] < budget['simultaneous_factor_storage_bytes']:
        raise ValueError('Memory budget cannot be smaller than simultaneous factors')
    if budget['peak_additional_bytes'] < budget['measured_peak_rss_bytes']:
        raise ValueError('Memory forecast below measured reference peak')
    try:
        available = int(current['memory_available'].split()[0])*1024
    except (KeyError, TypeError, ValueError):
        raise ValueError('Host available memory is not measurable') from None
    if current.get('memory_limit') is not None and current.get('memory_current') is not None:
        try:
            limit=int(current['memory_limit']); used=int(current['memory_current'])
        except (TypeError,ValueError):
            raise ValueError('Cgroup memory values are not finite limits') from None
        mode='CGROUP_AND_HOST'
    else:
        limits=current.get('address_space_rlimit',[])
        used=current.get('process_vm_size_bytes')
        peak_vm=budget.get('measured_peak_vm_bytes')
        if (current.get('cgroup_visibility') != 'NOT_MOUNTED_IN_JOB_NAMESPACE' or len(limits)!=2
                or any(type(x) is not int or x<=0 for x in limits) or limits[0]!=limits[1]
                or type(used) is not int or used<=0 or not isinstance(peak_vm,(int,float))
                or not np.isfinite(peak_vm) or peak_vm<=0 or budget['peak_additional_bytes']<peak_vm):
            raise ValueError('Missing finite enforced address-space limit and measured virtual-memory budget')
        limit=limits[0];mode='HARD_RLIMIT_AS_AND_HOST_CGROUP_UNOBSERVED'
    remaining = min(limit-used, available)
    needed = budget['peak_additional_bytes']+budget['reserve_bytes']
    if needed > remaining:
        raise MemoryError(f'Memory budget needs {needed} bytes, available {remaining}')
    return dict(needed_bytes=needed, available_bytes=remaining, status='PASS',resource_mode=mode,
                caveat='Unknown ancestor cgroup cannot be measured inside namespace' if mode.startswith('HARD') else None)

def compare_environment(reference, current):
    required = ('python', 'platform', 'machine', 'cpu_model', 'logical_cpus', 'affinity',
                'numpy', 'scipy', 'threadpools', 'requested_threads', 'blas_library_identity',
                'cgroup_visibility', 'address_space_rlimit')
    missing = [k for k in required if reference.get(k) is None or current.get(k) is None]
    differences = [k for k in required if reference.get(k) != current.get(k)]
    for k in ('cpu_quota','memory_limit'):
        if reference.get(k)!=current.get(k):differences.append(k)
    for value in (reference,current):
        if value.get('cgroup_visibility')=='NOT_MOUNTED_IN_JOB_NAMESPACE':
            limits=value.get('address_space_rlimit',[])
            if len(limits)!=2 or any(type(x) is not int or x<=0 for x in limits) or limits[0]!=limits[1]:missing.append('enforced_address_space_limit')
        elif value.get('cpu_quota') is None or value.get('memory_limit') is None:
            missing.append('cgroup_limits')
        if not value.get('blas_library_identity'):missing.append('blas_library_identity')
    return dict(comparable=not missing and not differences, missing=missing, differences=differences,
                note='Matching manifest does not eliminate unobserved host/cgroup contention; inspect runtime deviations')


def runtime_gate(forecast, elapsed, expected_counts, actual_counts, environment_match, calibrated):
    complete = all(k in forecast and np.isfinite(forecast[k]) for k in ('expected_seconds', 'lower_seconds', 'upper_seconds'))
    valid = complete and 0 < forecast['expected_seconds'] and 0 <= forecast['lower_seconds'] <= forecast['expected_seconds'] <= forecast['upper_seconds']
    if not valid:
        raise ValueError('Invalid declared runtime forecast')
    if not calibrated or not environment_match:
        status = 'NOT_EVALUABLE'
    elif expected_counts != actual_counts or not forecast['lower_seconds'] <= elapsed <= forecast['upper_seconds']:
        status = 'REVIEW_REQUIRED'
    else:
        status = 'PASS'
    return dict(**forecast, actual_seconds=elapsed, actual_over_expected=elapsed/forecast['expected_seconds'],
                deviation_seconds=elapsed-forecast['expected_seconds'], status=status,
                expected_counts=expected_counts, actual_counts=actual_counts,
                environment_comparable=environment_match, calibrated=calibrated,
                scope='Worker internal measured interval; enclosing process duration must be checked separately')

def digest(data):
    return hashlib.sha256(data).hexdigest()


def json_bytes(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()


def atomic_private(path, value):
    temp = path.with_name('.' + path.name + '-' + secrets.token_hex(8))
    with temp.open('xb') as handle:
        handle.write(json_bytes(value))
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temp, path)
    fd = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def safe_parts(path):
    p = PurePosixPath(path)
    if not path or p.is_absolute() or str(p) != path or any(
        x in ('.', '..') or x.startswith('.') or len(x) > 200 or
        any(ord(c) < 32 for c in x) for x in p.parts
    ):
        raise ValueError('Unsafe relative path')
    return p.parts


class Workspace:
    """All shared filesystem traversal uses directory FDs and refuses symlinks."""
    def __init__(self, path):
        self.path = Path(path)
        self.fd = os.open(path, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)

    @contextlib.contextmanager
    def directory(self, parts, create=False):
        fd = os.dup(self.fd)
        try:
            for part in parts:
                if create:
                    try:
                        os.mkdir(part, 0o750, dir_fd=fd)
                    except FileExistsError:
                        pass
                nxt = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                os.close(fd)
                fd = nxt
            yield fd
        finally:
            os.close(fd)

    def read(self, path, limit=MAX_FILE):
        parts = safe_parts(path)
        with self.directory(parts[:-1]) as parent:
            fd = os.open(parts[-1], os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=parent)
        with os.fdopen(fd, 'rb') as handle:
            before = os.fstat(handle.fileno())
            if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or before.st_size > limit:
                raise ValueError('Expected bounded regular, non-hardlinked file: ' + path)
            data = handle.read(limit + 1)
            after = os.fstat(handle.fileno())
        if len(data) > limit or (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
            raise ValueError('File changed or exceeds limit: ' + path)
        return data

    def write(self, path, data, replace=True):
        parts = safe_parts(path)
        with self.directory(parts[:-1], create=True) as parent:
            if not replace:
                try:
                    if self.read(path) != data:
                        raise ValueError('Existing input differs: ' + path)
                    return
                except FileNotFoundError:
                    pass
            temp = '.github-' + secrets.token_hex(12)
            fd = os.open(temp, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o640, dir_fd=parent)
            try:
                with os.fdopen(fd, 'wb') as handle:
                    os.fchmod(handle.fileno(), 0o640)
                    handle.write(data)
                    handle.flush()
                    os.fsync(handle.fileno())
                os.replace(temp, parts[-1], src_dir_fd=parent, dst_dir_fd=parent)
                os.fsync(parent)
            finally:
                try:
                    os.unlink(temp, dir_fd=parent)
                except FileNotFoundError:
                    pass

    def names(self, path):
        with self.directory(safe_parts(path)) as fd:
            return sorted(os.listdir(fd))

    def files(self, path):
        def walk(fd, prefix, depth):
            if depth > 16:
                raise ValueError('Too many nested result directories')
            for name in sorted(os.listdir(fd)):
                if name.startswith('.'):
                    continue
                safe_parts(name)
                info = os.stat(name, dir_fd=fd, follow_symlinks=False)
                relative = prefix + name
                if stat.S_ISDIR(info.st_mode):
                    child = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                    try:
                        yield from walk(child, relative + '/', depth + 1)
                    finally:
                        os.close(child)
                else:
                    yield relative
        with self.directory(safe_parts(path)) as root:
            yield from walk(root, '', 0)

    def discover(self):

            if not name.endswith('.json') or not ID.fullmatch(name[:-5]):
                continue
            path = self.records / name

                continue

                request = json.loads(self.work.read(self.queue_dir + '/' + name, 4096))
                if not isinstance(request, dict) or set(request) != {'run_id', 'commit'} or request['run_id'] != name[:-5] or not isinstance(request['commit'], str) or not SHA.fullmatch(request['commit']):
                    raise ValueError('Expected run_id and a full 40-character commit SHA')
            except (OSError, ValueError) as exc:
                self.save({'run_id': name[:-5], 'stage': 'rejected', 'last_error': str(exc)[:1000]})
                continue
            self.save({'run_id': request['run_id'], 'source_commit': request['commit'],
                       'repository': self.config['repository'], 'stage': 'accepted', 'created': time.time()})

    def prepare(self, record):
        rid, commit = record['run_id'], record['source_commit']
        self.git('fetch', '--no-tags', '--depth=1', 'origin', commit)
        actual = self.git('rev-parse', 'FETCH_HEAD').decode().strip()
        if actual != commit:
            raise ValueError('Fetched commit differs from request')
        # No checkout, hooks, attributes, submodules or repository code on the networked host.
        tree = self.git('ls-tree', '-rz', '--full-tree', commit)
        allowed = []
        total = 0
        for item in tree.split(b'\0'):
            if not item:
                continue
            metadata, name = item.split(b'\t', 1)
            name = name.decode('utf-8')
            if name.split('/')[0] in {'infra', 'results'} or any(p.startswith('.') for p in PurePosixPath(name).parts):
                continue
            safe_parts(name)
            mode, kind, oid = metadata.decode().split()
            if kind != 'blob' or mode not in {'100644', '100755'}:
                raise ValueError('Source symlinks and submodules are unsupported')
            size = int(self.git('cat-file', '-s', oid))
            total += size
            if size > MAX_FILE or total > MAX_TOTAL or len(allowed) >= MAX_FILES:

    def queue_job(self, rid, wrapper):
        identity = {'source_sha256': digest(wrapper), 'args': []}
        try:
            old = json.loads(self.work.read('jobs/' + rid + '/request.json', 4096))
            if old != identity:
                raise ValueError('Job id already used for different code')
            return
        except FileNotFoundError:
            pass
        health = json.loads(self.work.read('worker-health.json', 4096))
        if time.time() - health.get('heartbeat', 0) > 15:
            raise RuntimeError('Existing worker is not responding')
        if shutil.disk_usage(self.work.path).free < 768 * 1024**2:
            raise RuntimeError('Less than 768 MiB free disk')
        with self.work.directory(('jobs',)) as jobs:
            temp = '.github-' + secrets.token_hex(12)
            os.mkdir(temp, 0o770, dir_fd=jobs)
            fd = os.open(temp, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=jobs)
            try:
                os.fchmod(fd, 0o770)
                contents = {
                    'main.py': wrapper, 'request.json': json_bytes(identity),
                    'status.json': json_bytes({'state': 'queued', 'created': time.time(), 'source_sha256': digest(wrapper)}),
                }
                for name, data in contents.items():
                    out = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o660, dir_fd=fd)
                    with os.fdopen(out, 'wb') as handle:
                        os.fchmod(handle.fileno(), 0o660)
                        handle.write(data)
                        handle.flush()
                        os.fsync(handle.fileno())
                os.fsync(fd)
                os.rename(temp, rid, src_dir_fd=jobs, dst_dir_fd=jobs)
                os.fsync(jobs)
            finally:
                os.close(fd)

    def export_log(self, path):
        parts = safe_parts(path)
        with self.work.directory(parts[:-1]) as parent:
            fd = os.open(parts[-1], os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=parent)
        with os.fdopen(fd, 'rb') as handle:
            before = os.fstat(handle.fileno())
            if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1:
                raise ValueError('Log must be a regular non-hardlinked file')
            size = before.st_size
            if size <= MAX_FILE:
                data = handle.read(MAX_FILE + 1)
                detail = {'method': 'full', 'original_bytes': size, 'omitted_bytes': 0}
            else:
                part = MAX_FILE // 4
                first = handle.read(part)
                handle.seek(size - part)
                last = handle.read(part)
                omitted = size - len(first) - len(last)
                data = first + ('\n[... %d log bytes omitted; full log retained on VM ...]\n' % omitted).encode() + last
                detail = {'method': 'head_tail', 'original_bytes': size, 'head_bytes': len(first),
                          'tail_bytes': len(last), 'omitted_bytes': omitted}
            after = os.fstat(handle.fileno())
            if (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
                raise ValueError('Log changed while collecting')
        return data, detail

    def publish_files(self, files, message):
        self.git('fetch', '--no-tags', '--depth=1', 'origin', 'refs/heads/main')
        base = self.git('rev-parse', 'FETCH_HEAD').decode().strip()
        index = self.state / ('index-' + secrets.token_hex(8))
        environment = {'GIT_INDEX_FILE': str(index)}
        try:
            self.git('read-tree', base, extra_env=environment)
            for name, data in files.items():
                safe_parts(name)
                oid = self.git('hash-object', '-w', '--stdin', data=data).decode().strip()
                self.git('update-index', '--add', '--cacheinfo', '100644', oid,
                         name, extra_env=environment)
            tree = self.git('write-tree', extra_env=environment).decode().strip()
            if tree == self.git('rev-parse', base + '^{tree}').decode().strip():
                commit = base  # Previous push succeeded before the process stopped.
            else:

                                  data=(message + '\n').encode()).decode().strip()
                self.git('push', 'origin', commit + ':refs/heads/main')  # Never force-push.
            return commit
        finally:
            index.unlink(missing_ok=True)

            except (OSError, ValueError, RuntimeError, TypeError, KeyError, subprocess.SubprocessError) as exc:

                self.save(record)

"""Accept execution reports only through a supplied GitHub file reader.

There is intentionally no VM client, filesystem reader, or fallback transport.
The production adapter calls the GitHub connector for each path at result_commit.
Local integration tests substitute a separate bare Git repository for GitHub.
"""
import hashlib
import json
import re


class UnconfirmedResult(RuntimeError):
    """No complete, matching report has been verified in GitHub."""


def _relative(path):
    return isinstance(path, str) and bool(path) and all(
        part and not part.startswith('.') and not any(ord(c) < 32 for c in part)
        for part in path.split('/')
    )


def read_github_result(fetch_file, *, repository, result_commit, run_id, source_commit):
    """fetch_file(repository, path, ref) -> bytes; every read pins the same ref.

    Return a verified report, including artifacts obtained from GitHub. A valid
    failed/cancelled/interrupted execution is a confirmed report, not a success.
    Missing files, wrong versions, changed bytes and incomplete exports fail closed.
    """
    if not re.fullmatch(r'[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+', repository):
        raise UnconfirmedResult('Invalid repository')
    if not re.fullmatch(r'[A-Za-z0-9_-]{12,64}', run_id):
        raise UnconfirmedResult('Invalid run id')
    for value in (source_commit, result_commit):
        if not re.fullmatch(r'[0-9a-f]{40}', value):
            raise UnconfirmedResult('A full immutable commit SHA is required')
    prefix = 'results/' + run_id + '/'

    def get(path):
        try:
            data = fetch_file(repository, path, result_commit)
        except Exception as exc:
            raise UnconfirmedResult('GitHub report is unavailable: ' + path) from exc
        if not isinstance(data, bytes):
            raise UnconfirmedResult('GitHub adapter must return exact bytes')
        return data

    try:
        report = json.loads(get(prefix + 'run.json'))
    except (ValueError, UnicodeError) as exc:
        raise UnconfirmedResult('Invalid GitHub report JSON') from exc
    if not isinstance(report, dict) or any(report.get(key) != value for key, value in {
        'repository': repository, 'run_id': run_id, 'source_commit': source_commit,
    }.items()):
        raise UnconfirmedResult('Report does not match the requested repository, run or code version')
    if report.get('export_errors') != []:
        raise UnconfirmedResult('GitHub export is incomplete; inspect export_errors in GitHub')
    job = report.get('job')
    if not isinstance(job, dict) or job.get('state') not in {'completed', 'failed', 'cancelled', 'interrupted'}:
        raise UnconfirmedResult('Report has no terminal execution state')
    if job['state'] == 'completed' and (type(job.get('exit_code')) is not int or job['exit_code'] != 0):
        raise UnconfirmedResult('Completed report has an invalid exit code')
    files = report.get('files')
    if not isinstance(files, dict) or 'output.log' not in files or len(files) > 512:
        raise UnconfirmedResult('Report has no exported log')
    if report.get('report_profile') != 'medium-v1':
        raise UnconfirmedResult('Unknown report profile')
    if job['state'] == 'completed' and not {
        'files/README.md', 'files/metrics.json', 'files/data_manifest.json'
    } <= files.keys():
        raise UnconfirmedResult('Successful execution lacks the required detailed report')
    artifacts = {}
    total = 0
    for name, expected in files.items():
        if not _relative(name) or (name != 'output.log' and not name.startswith('files/')):
            raise UnconfirmedResult('Artifact path escapes the result directory')
        if not isinstance(expected, dict) or type(expected.get('bytes')) is not int or not 0 <= expected['bytes'] <= 8 * 1024**2:
            raise UnconfirmedResult('Invalid artifact metadata')
        total += expected['bytes']
        if total > 32 * 1024**2:
            raise UnconfirmedResult('Artifact set exceeds the supported export size')
        data = get(prefix + name)
        if len(data) != expected['bytes'] or hashlib.sha256(data).hexdigest() != expected.get('sha256'):
            raise UnconfirmedResult('GitHub artifact size/hash mismatch: ' + name)
        artifacts[name] = data
    return {
        'result_source': 'github', 'repository': repository, 'result_commit': result_commit,
        'source_commit': source_commit, 'run_id': run_id, 'report': report,
        'execution_succeeded': job['state'] == 'completed' and job['exit_code'] == 0,
        'artifacts': artifacts,
    }

           ('up_E',sp.csr_matrix(np.sqrt(.35)*np.kron(I,lower.T)))]
    H=sp.csr_matrix(H);L=build_liouvillian(H,jumps);space=Space(np.zeros(4,int))
    # space.pack uses the single all-operator block with standard Fortran vec.
    F=sp.csr_matrix(np.kron(n,I));obs=dict(F=F,J_h=F,source=F,h6=.1*F,P_m6_N2=.2*sp.eye(4),inventory=sp.eye(4))
    model=dict(H=H,L=L,jumps=jumps,space=space,observables=obs,dimensions=(2,2),case=dict(model=name,id='toy_'+name))
    A=L.toarray().copy();b=np.zeros(16,complex);A[-1]=space.trace_row.toarray();b[-1]=1
    rho=np.linalg.solve(A,b);raw=measure_raw(model,rho)
    evidence=dict(baseline=dict(raw=raw),corr=dict(result=dict(signed_area=.02,area_uncertainty=1e-7)))
    return model,rho,evidence


class ResetAlgebra(unittest.TestCase):
    def test_coherent_target_against_kraus_and_dense_superoperator(self):
        model,_,_=toy();tau=np.array([[.6,.12j],[-.12j,.4]]);R=Reset(model['space'],tau,(2,2))
        rng=np.random.default_rng(813);x=rng.normal(size=16)+1j*rng.normal(size=16);rho=model['space'].unpack(x)
        vals,vecs=np.linalg.eigh(tau);out=np.zeros((4,4),complex);complete=np.zeros((4,4),complex)
        for a in range(2):
            for b in range(2):
                K=np.kron(np.eye(2),np.sqrt(vals[a])*np.outer(vecs[:,a],vecs[:,b].conj()))
                out+=K@rho@K.conj().T;complete+=K.conj().T@K
        np.testing.assert_allclose(R.apply(x),model['space'].pack(out),atol=1e-13)
        np.testing.assert_allclose(complete,np.eye(4),atol=1e-13)
        np.testing.assert_allclose(R.apply(2*x+1j*x),2*R.apply(x)+1j*R.apply(x),atol=1e-13)
        self.assertEqual(R.rank,4);R.certify_action(x)

    def test_nontrivial_node_charge_rank_and_no_projection(self):
        s=Space(np.repeat([0,1,1,2],2));R=Reset(s,np.array([[.5,.2],[.2,.5]]),(4,2))
        self.assertEqual(R.rank,6)
        x=random_rhs(s,17);R.certify_action(x)
        rho=s.unpack(x);rho[0,7]=.01
        with self.assertRaises(ValueError):s.pack(rho)

    def test_invalid_target(self):
        for a in (np.eye(2),np.array([[1.2,0],[0,-.2]]),np.array([[.5,1],[0,.5]]),np.array([[np.nan]])):
            with self.assertRaises((ValueError,ArithmeticError)):target_check(a)

    def test_target_is_immutable(self):
        m,_,_=toy();tau=np.eye(2)/2;R=Reset(m['space'],tau,(2,2));tau[0,0]=0
        self.assertEqual(R.tau[0,0],.5)
        with self.assertRaises(ValueError):R.tau[0,0]=0

    def test_poisson_finite_rate_exact_series(self):
        # Pure reset generator is exactly an average over Poisson event counts.
        m,_,_=toy();R=Reset(m['space'],np.diag([.7,.3]),(2,2));x=random_rhs(m['space'],9)
        gamma=.4;t=.8;mat=(R.U@R.V).toarray();power=np.eye(16,dtype=complex);poisson=np.zeros((16,16),complex)
        weight=np.exp(-gamma*t)
        for n in range(24):
            poisson+=weight*power;power=mat@power;weight*=gamma*t/(n+1)
        np.testing.assert_allclose(poisson@x,expm(gamma*t*(mat-np.eye(16)))@x,atol=1e-13)


class ResetSolvers(unittest.TestCase):
    def test_woodbury_and_adjoint_match_dense(self):
        m,rho,_=toy();R=Reset(m['space'],np.array([[.6,.1j],[-.1j,.4]]),(2,2));rhs=random_rhs(m['space'],66)
        for method in ('lu','gmres_ilu'):
            s=Solver(m['L'],m['space'],.2,R,method=method)
            x,eq=s.solve(rhs);dense=s.base.toarray()+.2*(s.U@s.V).toarray()
            truth=np.linalg.solve(dense,np.r_[rhs,0.])[:-1]
            np.testing.assert_allclose(x,truth,atol=1e-10)
            if method=='lu':
                b=np.r_[rhs,0.];np.testing.assert_allclose(s.inverse(b,True),np.linalg.solve(dense.conj().T,b),atol=1e-12)
                self.assertGreater(s.condition_estimate()['value'],0)
            self.assertGreaterEqual(s.counts['factor_rhs_columns'],R.rank+1)
            self.assertLess(eq['absolute_inf'],1e-10)

    def test_response_sign_integral_and_finite_difference(self):
        from e002e_corr_exp.core import marginals
        m,rho,_=toy();s=Solver(m['L'],m['space']);R=Reset(m['space'],marginals(m['space'].unpack(rho),(2,2))[1],(2,2))
        q=R.apply(rho)-rho;X,_=s.solve(-q)
        # Exact augmented exponential integral, independent of inverse formula.
        A=np.zeros((17,17),complex);A[:16,:16]=m['L'].toarray();A[:16,16]=q
        integral=expm(A*100)[:16,16]
        np.testing.assert_allclose(X,integral,atol=1e-11)
        eps=1e-5;shifted=Solver(m['L'],m['space'],eps,R);new,_=shifted.solve(trace=1.)
        np.testing.assert_allclose((new-rho)/eps,X,rtol=1e-4,atol=1e-6)

    def test_zero_response_full_reset_control(self):
        m,rho,_=toy();q=rho*m['space'].trace(rho)-rho
        X,eq=Solver(m['L'],m['space']).solve(-q)
        self.assertLess(maxabs(X),1e-12)
        _,eq=Solver(m['L'],m['space']).solve(np.zeros(16))
        self.assertIsNone(eq['relative_true'])

    def test_traceful_rhs_rejected(self):
        m,rho,_=toy()
        with self.assertRaises(ValueError):Solver(m['L'],m['space']).solve(rho)

    def test_gmres_failure_not_pass(self):
        m,_,_=toy()
        with patch('e002e_reset_exp.numerics.gmres',return_value=(np.zeros(17),7)):
            with self.assertRaises(ArithmeticError):Solver(m['L'],m['space'],method='gmres_ilu').solve(trace=1.)


class Decisions(unittest.TestCase):
    def test_sign_boundaries(self):
        for s,u,expected in [(1,0,'SUPPORTED'),(-1,0,'REFUTED'),(1,1,'UNRESOLVED'),(0,0,'UNRESOLVED')]:
            self.assertEqual(sign_decision(s,u)['status'],expected)

    def test_relative_intervals_and_guard(self):
        self.assertEqual(relative_decision(1.01,.001,1,.001,1,.001,.1)['status'],'SUPPORTED')
        self.assertEqual(relative_decision(1.2,.001,1,.001,1,.001,.1)['status'],'REFUTED')
        self.assertEqual(relative_decision(1.1,.02,1,.02,1,.02,.1)['status'],'UNRESOLVED')
        self.assertEqual(relative_decision(0,0,0,0,0,0,.1)['status'],'NOT_ASSESSED_SMALL_SIGNAL')

    def test_small_S_does_not_disable_window_or_sign(self):
        a=dict(S=5e-7,flux={'0.0':.1,'0.001':.1000000005,'0.002':.100000001})
        r=analyze(a,a,.05,1e-6)
        self.assertEqual(r['sign']['status'],'SUPPORTED')
        self.assertEqual(r['linearity_status'],'NOT_ASSESSED_SMALL_SIGNAL')
        self.assertEqual(r['window']['status'],'REFUTED')

    def test_caps_and_nonfinite_block(self):
        a=dict(S=.05,flux={'0.0':.1,'0.001':.10005,'0.002':.1001})
        with self.assertRaises(ArithmeticError):analyze(a,dict(a,S=.051),.05,1e-6)
        b=dict(a,flux=dict(a['flux'],**{'0.001':.10006}))
        with self.assertRaises(ArithmeticError):analyze(a,b,.05,1e-6)
        with self.assertRaises(ValueError):sign_decision(float('nan'),0)
