"""Explicit diagnostics and predeclared prediction evaluation; no fitted labels."""
import numpy as np
from scipy import sparse as sp
from scipy.sparse.linalg import splu
from e002_exp.model import maxabs
from .numerics import FLOOR, ratio


def renewal_check(model, state, tolerance=1e-9):
    if model.get('original_dimension') != 4:
        return dict(status='NOT_APPLICABLE')
    space = model['space']; groups = [i for i, g in enumerate(space.groups) if space.charges[g[0]] == 1]
    if len(groups) != 1:
        return dict(status='UNRESOLVED_EXCITATION_BLOCK')
    group = groups[0]; lo, hi = space.offsets[group:group+2]
    # Injection state is generated by the physical source from the empty state.
    C = dict(model['jumps'])['source_0']; empty = np.flatnonzero(space.charges == 0)
    if len(empty) != 1:
        return dict(status='UNRESOLVED_EMPTY_BRANCH')
    rho0 = np.zeros((space.d, space.d), complex); rho0[empty[0], empty[0]] = 1.
    injection = C@rho0@C.getH(); P = float(np.trace(injection).real)
    b = space.pack(injection/P)[lo:hi]; K = model['L'][lo:hi, lo:hi]
    try:
        R = splu(K.tocsc()).solve(-b)
    except RuntimeError:
        return dict(status='DARK_OR_SINGULAR_BLOCK_REQUIRES_BRANCH_ANALYSIS')
    full_R = np.zeros(space.size, complex); full_R[lo:hi] = R
    tau = float(space.trace(full_R).real)
    if tau <= 0 or maxabs(K@R+b) > tolerance:
        return dict(status='INVALID_RENEWAL_SOLUTION')
    predicted = P*full_R/(1+P*tau)
    ground = np.zeros((space.d, space.d), complex); ground[empty[0], empty[0]] = 1/(1+P*tau)
    predicted += space.pack(ground)
    error = maxabs(predicted-state)
    return dict(status='PASS' if error <= tolerance else 'FAIL', tau=tau, P_tau=P*tau,
                predicted_F=P/(1+P*tau), predicted_F_over_p_e=1/tau, state_error_inf=error)


def symmetry_check(model, tolerance=1e-12):
    """Test W K relating epsilon and -epsilon at Phi=pi/2, before using parity."""
    if model['space'].d != 216 or abs(model['case']['phi_over_pi']-.5) > 1e-12:
        return dict(symmetry_applicable=False, operator_covariance_residual=None,
                    parity_residual_max=None, parity_tolerance=None, parity_status='NOT_APPLICABLE')
    edges = np.array(list(np.ndindex(3, 3, 3)))
    diagonal = np.array([(-1.)**(((node >> 1) & 1)+sum(edge == 2)) for node in range(8) for edge in edges])
    W = sp.diags(diagonal); H = model['H']; epsilon = model['case']['epsilon']
    err = maxabs((H-2*epsilon*model['n'][1])+W@H.conjugate()@W)
    for _, C in model['jumps']:
        transformed = W@C.conjugate()@W
        # All declared jumps transform up to a real sign in this convention.
        err = max(err, min(maxabs(transformed-C), maxabs(transformed+C)))
    P = model['observables']['P_m6_N2']; J = model['observables']['h6']
    err = max(err, maxabs(W@P.conjugate()@W-P), maxabs(W@J.conjugate()@W-J))
    return dict(symmetry_applicable=err <= tolerance, operator_covariance_residual=err,
                parity_residual_max=None, parity_tolerance=None,
                parity_status='OPERATOR_CHECK_PASSED_STATE_PAIRS_PENDING' if err <= tolerance else 'OPERATOR_CHECK_FAILED')


def f2_differences(records, baseline):
    # Baseline is a pinned, previously verified report of undeformed cases.
    refs = {(r['case']['model'], r['case']['phi_over_pi'], r['case']['epsilon'], r['case']['node_noise']): r
            for r in baseline['records'] if r['case'].get('m6_v_over_J') == 1.}
    if baseline.get('status') != 'VERIFIED_GITHUB':
        raise ValueError('Unverified baseline')
    out = []
    for record in records:
        c = record['case']; key = (c['model'], c['phi_over_pi'], c['epsilon'], c['node_noise'])
        if key not in refs:
            raise ValueError('F2 lacks matching baseline')
        base = refs[key]; changes = {name: record['raw'][name]-base['raw'][name] for name in ('J_h', 'h6', 'P_m6_N2')}
        a, b = record['conditional']['j6'], base['conditional']['j6']
        changes['j6'] = None if a['value'] is None or b['value'] is None else a['value']-b['value']
        changes['j6_uncertainty'] = None if a['uncertainty'] is None or b['uncertainty'] is None else a['uncertainty']+b['uncertainty']
        out.append(dict(id=record['id'], baseline_id=base['id'], changes=changes))
    return out


def evaluate_predictions(predictions, records, differences):
    """Numerical verdict only when the requested metric and error are resolved."""
    records = {r['id']: r for r in records}; diffs = {r['id']: r for r in differences}; out = []
    for p in predictions['predictions']:
        result = dict(prediction_id=p['prediction_id'], status='EXPLORATORY')
        if p['confirmatory_or_exploratory'] == 'exploratory':
            out.append(result); continue
        definition = p['observable_definition']; name = definition['observable']; terms = definition['linear_terms']
        value = 0.; uncertainty = 0.; unresolved = False; missing_error = False
        for term in terms:
            case_id = term['case_id']; coefficient = float(term['coefficient']); r = records[case_id]
            if definition['kind'] == 'baseline_difference':
                d = diffs[case_id]['changes']; v = d.get(name); u = d.get(name+'_uncertainty')
            elif name in r['conditional']:
                v, u = r['conditional'][name]['value'], r['conditional'][name]['uncertainty']
            else:
                v = r['raw'].get(name); u = r.get('primitive_uncertainty', {}).get(name)
            if v is None:
                unresolved = True; continue
            value += coefficient*v
            if u is None:
                missing_error = True
            else:
                uncertainty += abs(coefficient)*u
        result.update(value=None if unresolved else float(value), uncertainty=None if missing_error else float(uncertainty))
        if unresolved or missing_error:
            result['status'] = 'NUMERICALLY_UNRESOLVED'; out.append(result); continue
        tol = p['tolerance_and_error_rule']['absolute_tolerance']; expected = p['expected_value_or_sign_or_interval']
        if p['prediction_type'] == 'value':
            lower, upper = float(expected)-tol, float(expected)+tol
        elif p['prediction_type'] == 'interval':
            lower, upper = float(expected[0])-tol, float(expected[1])+tol
        else:
            lower, upper = (tol, np.inf) if expected == 1 else (-np.inf, -tol)
        a, b = value-uncertainty, value+uncertainty
        result['status'] = 'SUPPORTED_WITHIN_DECLARED_TOLERANCE' if a >= lower and b <= upper else ('PREDICTION_FAILED' if b < lower or a > upper else 'NUMERICALLY_UNRESOLVED')
        out.append(result)
    return out

            raise ValueError('Nonfinite complex output')
        return dict(re=float(value.real), im=float(value.imag))
    if isinstance(value, np.ndarray):
        return serial(value.tolist())
    if isinstance(value, np.generic):
        return serial(value.item())
    if isinstance(value, dict):
        return {k: serial(v) for k, v in value.items()}
    if isinstance(value, (tuple, list)):
        return [serial(v) for v in value]
    return value


def real_values(values, tolerance=1e-9):
    out = {}
    for name, v in values.items():
        if name.startswith('C_'):
            out[name] = complex(v)
        elif abs(complex(v).imag) > tolerance:
            raise ArithmeticError('Hermitian observable has imaginary residue: '+name)
        else:
            out[name] = float(complex(v).real)
    return out


def conditional_values(raw, errors=None):
    errors = errors or {}
    def conditional(h, p):
        return ratio(raw[h], raw[p], errors.get(h), errors.get(p))
    out = {}
    for key in raw:
        if key.startswith('p_m') and '_N' in key:
            suffix = key[1:]; out['j'+suffix] = conditional('h'+suffix, key)
        if key.startswith('p_N'):
            out['j'+key[1:]] = conditional('h'+key[1:], key)
    if 'P_m6_N2' in raw:
        out['j6'] = conditional('h6', 'P_m6_N2')
    if 'p_e' in raw:
        out['F_over_p_e'] = conditional('F', 'p_e')
    return out


def validate_conditionals(raw, conditioned, errors=None):
    expected = conditional_values(raw, errors)
    if set(expected) != set(conditioned):
        raise ValueError('Conditional fields mismatch')
    for key, item in expected.items():
        actual = conditioned[key]['value']
        if (item['value'] is None) != (actual is None):
            raise ValueError('Conditional null policy mismatch')
        if actual is not None and (not np.isfinite(actual) or abs(actual-item['value']) > 1e-12*max(1., abs(actual))):
            raise ValueError('Conditional reconstruction mismatch')
        if conditioned[key].get('uncertainty') != item['uncertainty']:
            raise ValueError('Conditional uncertainty mismatch')

def node_ops():
    eye = sp.eye(8, dtype=complex, format='csr')
    def emb(x, at):
        return tensor([x if k == at else sp.eye(2, format='csr') for k in range(3)])
    up = [emb(ketbra(2, 1, 0), k) for k in range(3)]
    n = [x @ x.getH() for x in up]
    return eye, up, n


def make_small(case):
    d = int(case['dimension']); mask = int(case['mask'])
    if d not in (4, 8) or mask not in (6, 7):
        raise ValueError('Unsupported ladder case')
    source = float(case.get('source', .5)); sink = float(case.get('sink', .5))
    noise = float(case.get('node_noise', .05)); v = float(case.get('v_over_J', 1.))
    eps = float(case['epsilon']); phi = float(case['phi_over_pi'])*np.pi
    if min(source, sink) <= 0 or min(noise, v) < 0 or not np.isfinite([source, sink, noise, v, eps, phi]).all():
        raise ValueError('Invalid small-model parameters')
    eye, up8, n8 = node_ops()
    if d == 4:
        # Order g, node0, node1, node2; independent global N<=1 model.
        eye = sp.eye(4, format='csr', dtype=complex)
        up = [ketbra(4, i+1, 0) for i in range(3)]
        n = [ketbra(4, i+1, i+1) for i in range(3)]
        charges = np.array([0, 1, 1, 1])
        hops = [ketbra(4, i+1, j+1) for i, j in PAIRS]
    else:
        up, n = up8, n8
        charges = np.array([k.bit_count() for k in range(8)])
        hops = [up[i] @ up[j].getH() for i, j in PAIRS]
    H = eps*n[1]; currents = []

        amp = v if e == (1 if mask == 6 else 0) else 1.
        h = amp*(np.exp(1j*theta)*a + np.exp(-1j*theta)*a.getH()) if mask & (1 << e) else 0*a
        H += h; currents.append(clean(1j*(h@n[j]-n[j]@h)))
    jumps = [('source_0', np.sqrt(source)*up[0]), ('sink_2', np.sqrt(sink)*up[2].getH())]
    convention = case.get('jump_convention', 'three_sqrt_phi_Z')
    if convention == 'sqrt(phi)*n1_only':
        jumps.append(('node_dephase_1', np.sqrt(noise)*n[1]))
    elif convention == 'three_sqrt_phi_Z':
        jumps += [(f'node_dephase_{i}', np.sqrt(noise)*(eye-2*x)) for i, x in enumerate(n)]
    else:
        raise ValueError('Unknown small noise convention')
    active = np.arange(d)
    branch = 'empty_initial_state_unique_stationary_candidate'
    if mask == 6 and v == 0:
        active = np.flatnonzero(abs(n[1].diagonal()) < .5)
        branch = 'n1_zero_invariant_from_empty_initial_state'
        outside = np.setdiff1d(np.arange(d), active)
        if maxabs(H[outside][:, active]) > 1e-12 or any(maxabs(c[outside][:, active]) > 1e-12 for _, c in jumps):
            raise ValueError('Declared physical branch is not invariant')
    restrict = lambda a: clean(a[active][:, active])
    H = restrict(H); jumps = [(name, restrict(c)) for name, c in jumps]
    n = [restrict(x) for x in n]; currents = [restrict(x) for x in currents]
    space = Space(charges[active]); charges = charges[active]
    obs = {'J_h': currents[1]+currents[2], 'F': sink*n[2], 'p_e': sp.diags((charges == 1).astype(float)),
           'P_N2': sp.diags((charges == 2).astype(float)), 'K': currents[1]-currents[2],
           'source': source*(sp.eye(len(active))-sum(n)) if d == 4 else source*(sp.eye(len(active))-n[0])}
    for N in range(4):
        p = sp.diags((charges == N).astype(float), format='csr')
        obs[f'p_N{N}'] = p; obs[f'h_N{N}'] = clean(p@obs['J_h'])

    currents = []; deformation = 0*H; hop_terms = []

def selective_difference(space, target):

"""Bordered sparse solves; state and response checks deliberately differ."""
import time
import numpy as np
from scipy import sparse as sp
from scipy.sparse.linalg import splu, spilu, gmres, LinearOperator, onenormest
from e002_exp.model import maxabs

FLOOR = 1e-6


def residual(A, x, b):
    r = A@x-b
    return dict(absolute_inf=maxabs(r), relative_2=float(np.linalg.norm(r)/max(np.linalg.norm(b), 1.)))


class Bordered:
    def __init__(self, L, space, z=0j, method='lu', rtol=1e-11, atol=1e-13,
                 maxiter=400, restart=100, ilu_drop_tol=1e-5, ilu_fill_factor=10.):
        self.space = space; self.L = sp.csr_matrix(L); self.z = complex(z)
        self.A = self.L if z == 0 else (z*sp.eye(space.size, format='csr')-self.L)
        r = space.trace_row.T/space.d
        self.matrix = sp.bmat([[self.A, r], [space.trace_row, None]], format='csc')
        self.rtol, self.atol, self.maxiter, self.restart = rtol, atol, maxiter, restart
        self.method = method; self.factor = None; self.preconditioner = None
        self.iterations = 0; self.solves = 0; self.solve_seconds = 0.
        start = time.perf_counter()
        if method == 'lu':
            self.factor = splu(self.matrix)
        elif method == 'gmres_ilu':
            self.factor = spilu(self.matrix, drop_tol=ilu_drop_tol, fill_factor=ilu_fill_factor)
            self.preconditioner = LinearOperator(self.matrix.shape, matvec=self.factor.solve, dtype=complex)
        else:
            raise ValueError('Unsupported solver')
        self.factor_seconds = time.perf_counter()-start
        self.factor_nnz = int(self.factor.L.nnz+self.factor.U.nnz)
        self.factor_storage_bytes = int(sum(a.data.nbytes+a.indices.nbytes+a.indptr.nbytes for a in (self.factor.L, self.factor.U)))

    def solve(self, b=None, trace=0.):
        b = np.zeros(self.space.size, complex) if b is None else np.asarray(b, complex)
        if b.shape != (self.space.size,) or not np.isfinite(b).all():
            raise ValueError('Invalid RHS')
        if abs(self.space.trace(b)) > max(1e-11, 10*self.atol):
            raise ValueError('RHS must have zero trace')
        if self.z != 0 and trace != 0:
            raise ValueError('Nonzero frequency expects traceless response')
        rhs = np.r_[b, trace]; start = time.perf_counter(); iterations = 0
        if self.method == 'lu':
            x = self.factor.solve(rhs)
            # Residual correction, never a state projection or renormalization.
            for _ in range(2):
                correction = self.factor.solve(rhs-self.matrix@x)
                x += correction
        else:
            def count(_):
                nonlocal iterations
                iterations += 1
            x, info = gmres(self.matrix, rhs, M=self.preconditioner, rtol=self.rtol,
                            atol=self.atol, maxiter=self.maxiter, restart=self.restart,
                            callback=count, callback_type='pr_norm')
            if info != 0:
                raise ArithmeticError('GMRES did not converge: '+str(info))
        self.solve_seconds += time.perf_counter()-start; self.solves += 1; self.iterations += iterations
        checks = residual(self.A, x[:-1], b)
        checks.update(trace_error=float(abs(self.space.trace(x[:-1])-trace)), border_multiplier=float(abs(x[-1])), iterations=iterations)
        limit = max(10*self.atol, 10*self.rtol*max(1., np.linalg.norm(b)))
        if not np.isfinite(x).all() or max(checks['absolute_inf'], checks['trace_error'], checks['border_multiplier']) > limit:
            raise ArithmeticError('Original equation/trace residual failed: '+str(checks))
        return x[:-1], checks

    def condition_estimate(self):
        if self.method != 'lu':
            return dict(value=None, status='NOT_ESTIMATED_FOR_ITERATIVE_SOLVER')
        inverse = LinearOperator(self.matrix.shape, matvec=self.factor.solve,
                                 rmatvec=lambda x: self.factor.solve(x, trans='H'), dtype=complex)
        estimate = float(sp.linalg.norm(self.matrix, 1)*onenormest(inverse))
        return dict(value=estimate, status='ONE_NORM_ESTIMATE_NOT_RIGOROUS_ERROR_BOUND')

    def metrics(self):
        return dict(method=self.method, matrix_nnz=int(self.matrix.nnz), factor_nnz=self.factor_nnz,
                    factor_storage_bytes=self.factor_storage_bytes, factor_seconds=self.factor_seconds,
                    solves=self.solves, solve_seconds=self.solve_seconds, iterations=self.iterations)


def state_checks(L, space, x, tolerance=1e-10):
    rho = space.unpack(x)
    eigmin = min(float(np.linalg.eigvalsh((rho[np.ix_(g, g)]+rho[np.ix_(g, g)].conj().T)/2).min()) for g in space.groups)
    result = dict(trace_error=float(abs(space.trace(x)-1)), hermiticity=maxabs(rho-rho.conj().T),
                  minimum_eigenvalue=eigmin, Lrho_inf=maxabs(L@x), Lrho_2=float(np.linalg.norm(L@x)))
    result['pass'] = bool(result['trace_error'] <= tolerance and result['hermiticity'] <= tolerance
                          and eigmin >= -tolerance and result['Lrho_inf'] <= tolerance)
    return result


def values(model, vector):
    return {name: complex(model['space'].row(op)@vector) for name, op in model['observables'].items()}


def ratio(h, p, uh=None, up=None):
    if not np.isfinite([h, p]).all() or p <= FLOOR:
        return dict(value=None, uncertainty=None, status='SMALL_DENOMINATOR')
    j = h/p
    if uh is None or up is None:
        return dict(value=float(j), uncertainty=None, status='ERROR_UNRESOLVED')
    if min(uh, up) < 0 or not np.isfinite([uh, up]).all():
        raise ValueError('Invalid error estimates')
    if p-up <= FLOOR:
        return dict(value=None, uncertainty=None, status='DENOMINATOR_UNRESOLVED')
    return dict(value=float(j), uncertainty=float((uh+abs(j)*up)/(p-up)), status='PROPAGATED_ESTIMATE')


def contrast(left, center, right, delta):
    if delta <= 0:
        raise ValueError('delta must be positive')
    if any(x['value'] is None for x in (left, center, right)):
        return dict(D=None, d=None, curvature=None, uncertainty_D=None, status='UNDEFINED')
    D = center['value']-(left['value']+right['value'])/2
    u = None if any(x['uncertainty'] is None for x in (left, center, right)) else center['uncertainty']+(left['uncertainty']+right['uncertainty'])/2
    return dict(D=float(D), d=None if abs(center['value']) <= FLOOR else float(-D/center['value']),
                curvature=float(-2*D/delta**2), uncertainty_D=u,
                center_below_both=bool(center['value'] < min(left['value'], right['value'])),
                status='ERROR_UNRESOLVED' if u is None else 'PROPAGATED_ESTIMATE')

def response_outputs(model, state, x):
    vals = values(model, x); base = values(model, state)
    if 'P_m6_N2' in vals:
        p = base['P_m6_N2'].real; h = base['h6'].real
        vals['j6'] = None if p <= FLOOR else (vals['h6']-(h/p)*vals['P_m6_N2'])/p

def static_response(model, state, target, prepared):
    V = selective_difference(model['space'], target)
    x, checks = prepared.solve(-(V@state))
    return x, response_outputs(model, state, x), checks


def source_input(model, state):
    # The jump already includes sqrt(kappa_in); D[C] = kappa_in D[sigma+].
    C = dict(model['jumps'])['source_0']
    return model['space'].pack(dissipator_action(C, model['space'].unpack(state)))


def frequency_response(model, state, omega, prepared_zero=None, **solver_options):
    b = source_input(model, state)
    solver = prepared_zero if omega == 0 and prepared_zero is not None else Bordered(model['L'], model['space'], z=1j*omega, **solver_options)
    x, checks = solver.solve(-b if omega == 0 else b)
    return x, response_outputs(model, state, x), checks, solver.metrics()


def qrt_inputs(model, state, names=('J_h', 'P_m6_N2', 'K')):
    space = model['space']; rho = space.unpack(state); base = values(model, state)
    inputs = []; variances = []
    for name in names:
        B = model['observables'][name]; mean = base[name]
        if abs(mean.imag) > 1e-10:
            raise ArithmeticError('Hermitian mean is complex')
        y = space.pack(B@rho-mean*rho)
        if abs(space.trace(y)) > 1e-10:
            raise ArithmeticError('QRT input trace failed')
        var = complex(space.row(B@B)@state-mean**2)
        if abs(var.imag) > 1e-10 or var.real < -1e-10:
            raise ArithmeticError('Invalid variance')
        inputs.append(y); variances.append(max(0., var.real))
    return np.column_stack(inputs), np.array(variances)


def propagate_qrt(model, state, horizon, dt, sample_interval, names=('J_h', 'P_m6_N2', 'K'), checkpoint=None):
    if min(horizon, dt, sample_interval) <= 0:
        raise ValueError('Invalid QRT grid')
    steps = round(horizon/dt); stride = round(sample_interval/dt)
    if stride < 1 or abs(steps*dt-horizon) > 1e-9 or abs(stride*dt-sample_interval) > 1e-9 or steps % stride:
        raise ValueError('QRT grid must have integral steps and common final sample')
    y, var = qrt_inputs(model, state, names); L = model['L']; space = model['space']
    rows = np.vstack([space.row(model['observables'][name]) for name in names])
    times = [0.]; out = [rows@y]; trace_max = maxabs(space.trace(y)); start = time.perf_counter()
    for step in range(1, steps+1):
        a = L@y; b = L@(y+dt*a/2); c = L@(y+dt*b/2); d = L@(y+dt*c)
        y += dt*(a+2*b+2*c+d)/6
        if step % stride == 0:
            if not np.isfinite(y).all():
                raise ArithmeticError('QRT became nonfinite')
            t = step*dt; times.append(t); out.append(rows@y)
            trace_max = max(trace_max, maxabs(space.trace(y)))
            if checkpoint:
                checkpoint(step, t, y)
    if trace_max > 1e-8:
        raise ArithmeticError('QRT trace drift: '+str(trace_max))
    times = np.array(times); out = np.array(out); normalized = np.full_like(out, np.nan)
    den = np.sqrt(var[:, None]*var[None, :]); valid = den > FLOOR
    normalized[:, valid] = out[:, valid]/den[valid]
    integral = np.trapezoid(abs(out.diagonal(axis1=1, axis2=2)), times, axis=0)
    tau = [float(a/v) if v > FLOOR else None for a, v in zip(integral, var)]
    late = out[int(.8*len(out)):]
    summary = dict(steps=steps, block_rhs_calls=4*steps, scalar_rhs_equivalents=4*steps*len(names),
                   inputs=len(names), outputs_per_input=len(names), samples=len(times), trace_max=trace_max,
                   elapsed_seconds=time.perf_counter()-start, variances=var.tolist(), tau_int_truncated=tau,
                   late_window_abs_max=float(abs(late).max()), tail_status='TAIL_UNRESOLVED',
                   infinite_integral=None, input_convention='(B-mean_B)rho; ordered connected QRT',
                   positivity_test_applied=False)
    return dict(times=times, C=out, normalized=normalized, normalization_valid=valid, summary=summary, final=y)


def compare_qrt(fine, coarse, atol, rtol):
    if not np.array_equal(fine['times'], coarse['times']):
        raise ValueError('QRT comparison grids differ')
    err = abs(fine['C']-coarse['C']); scale = atol+rtol*np.maximum(abs(fine['C']), abs(coarse['C']))
    return dict(pass_gate=bool(np.all(err <= scale)), max_abs=float(err.max()),
                max_scaled=float((err/scale).max()), uncertainty_kind='EMPIRICAL_DT_DIFFERENCE_NOT_RIGOROUS_BOUND')


def schur_response(L, b, row, target, z, rtol=1e-10, maxiter=300):
    """Complete T/B partition; matrix-free Schur complement, no dense inverse.

    This candidate decomposition alone cannot certify Fano. z=0 is deliberately
    rejected here: stationary constraints need the separately validated border.
    """
    if z == 0:
        raise ValueError('Schur candidate at z=0 requires a separately declared constrained partition')
    L = sp.csr_matrix(L); n = L.shape[0]; target = np.asarray(target, dtype=int)
    if target.ndim != 1 or len(np.unique(target)) != len(target) or not 0 < len(target) < n or min(target) < 0 or max(target) >= n:
        raise ValueError('Invalid Schur partition')
    other = np.setdiff1d(np.arange(n), target); A = z*sp.eye(n, format='csr')-L
    tt = A[target][:, target]; tb = A[target][:, other]; bt = A[other][:, target]; bb = A[other][:, other]
    factor = splu(bb.tocsc()); rb_b = factor.solve(np.asarray(b)[other])
    eff = LinearOperator(tt.shape, matvec=lambda x: tt@x-tb@factor.solve(bt@x), dtype=complex)
    rhs = np.asarray(b)[target]-tb@rb_b
    xt, info = gmres(eff, rhs, rtol=rtol, atol=rtol*1e-2, maxiter=maxiter)
    if info:
        raise ArithmeticError('Schur GMRES did not converge')
    xb = factor.solve(np.asarray(b)[other]-bt@xt)
    x = np.empty(n, complex); x[target] = xt; x[other] = xb
    err = maxabs(A@x-b)
    if err > 10*rtol*max(1., np.linalg.norm(b)):
        raise ArithmeticError('Schur reconstruction residual failed')
    row = np.asarray(row); bg = complex(row[other]@rb_b)
    mediated = complex(row[target]@xt-row[other]@factor.solve(bt@xt))
    total = complex(row@x)
    return dict(x=x, background=bg, mediated=mediated, total=total,
                cross_intensity=float(2*np.real(bg.conjugate()*mediated)), reconstruction_error=float(abs(total-bg-mediated)),
                equation_residual=err, target_size=len(target), complement_size=len(other),
                interpretation='CANDIDATE_DECOMPOSITION_REQUIRES_PHYSICAL_PARTITION_AND_HELDOUT_CHECKS')


def g0(epsilon, v, J=1., kappa_left=.5, kappa_right=.5):
    A = J*J+kappa_left*kappa_right/4; width = kappa_left*v*v/(2*A)
    bg = -J*np.sqrt(kappa_left*kappa_right)/A
    if v == 0:
        amplitude, resonance = complex(bg), 0j
    else:
        amplitude = bg*epsilon/(epsilon-1j*width); resonance = amplitude-bg
    return dict(amplitude=amplitude, background=complex(bg), resonance=resonance,
                intensity=float(abs(amplitude)**2), width=width,
                cross=float(2*np.real(complex(bg).conjugate()*resonance)))

===== FILE e002e_exp/space.py =====
"""Charge-covariant q=0 Liouvillian, without a dense Liouville matrix."""
from dataclasses import dataclass
import numpy as np
from scipy import sparse as sp
from e002_exp.model import clean, maxabs


@dataclass
class Space:
    charges: np.ndarray

    def __post_init__(self):
        self.charges = np.asarray(self.charges, dtype=int)
        self.d = len(self.charges)
        self.groups = [np.flatnonzero(self.charges == n) for n in np.unique(self.charges)]
        self.sizes = [len(g) ** 2 for g in self.groups]
        self.offsets = np.r_[0, np.cumsum(self.sizes)]
        self.indices = np.concatenate([(g[:, None] + self.d*g[None, :]).ravel(order='F') for g in self.groups])
        self.trace_indices = np.concatenate([off + np.arange(len(g))*(len(g)+1)
                                            for off, g in zip(self.offsets, self.groups)])
        self.size = len(self.indices)
        self.trace_row = sp.csr_matrix((np.ones(self.d), (np.zeros(self.d, int), self.trace_indices)), shape=(1, self.size))

    def pack(self, matrix, tolerance=1e-12):
        a = matrix.toarray() if sp.issparse(matrix) else np.asarray(matrix)
        if a.shape != (self.d, self.d):
            raise ValueError('Hilbert dimensions disagree')
        if maxabs(np.where(self.charges[:, None] != self.charges[None, :], a, 0)) > tolerance:
            raise ValueError('Input has components outside q=0')
        return a.ravel(order='F')[self.indices].copy()

    def unpack(self, vector):
        v = np.asarray(vector)
        if v.shape != (self.size,):
            raise ValueError('Reduced vector shape mismatch')
        a = np.zeros(self.d*self.d, complex)
        a[self.indices] = v
        return a.reshape(self.d, self.d, order='F')

    def row(self, operator):
        # trace(O rho), without conjugation of O: vec(O.T)^T vec(rho).
        a = operator.toarray() if sp.issparse(operator) else np.asarray(operator)
        return a.T.ravel(order='F')[self.indices]

    def trace(self, vector):
        return np.asarray(vector)[self.trace_indices].sum(axis=0)


def charge_checks(H, jumps, space, tolerance=1e-12):
    h = sp.coo_matrix(H)
    hbad = maxabs(h.data[space.charges[h.row] != space.charges[h.col]])
    deltas = {}
    for name, C in jumps:
        c = sp.coo_matrix(C)
        nz = abs(c.data) > tolerance
        ds = np.unique(space.charges[c.row[nz]] - space.charges[c.col[nz]])
        if len(ds) > 1:
            raise ValueError('Jump lacks definite charge: ' + name)
        deltas[name] = int(ds[0]) if len(ds) else None
    if hbad > tolerance:
        raise ValueError('Hamiltonian violates charge covariance')
    return dict(H_charge_residual=hbad, jump_charge_changes=deltas)


def liouvillian(H, jumps, space):
    """Exact block construction; every within-charge coherence is retained."""
    charge_checks(H, jumps, space)
    H = sp.csr_matrix(H)
    count = len(space.groups)
    blocks = [[sp.csr_matrix((space.sizes[a], space.sizes[b]), dtype=complex)
               for b in range(count)] for a in range(count)]
    for a, g in enumerate(space.groups):
        h = H[g][:, g]; eye = sp.eye(len(g), format='csr')
        blocks[a][a] += -1j*(sp.kron(eye, h) - sp.kron(h.T, eye))
    for _, C in jumps:
        C = sp.csr_matrix(C); loss = clean(C.getH() @ C)
        for a, ga in enumerate(space.groups):
            eye = sp.eye(len(ga), format='csr'); m = loss[ga][:, ga]
            blocks[a][a] -= .5*(sp.kron(eye, m) + sp.kron(m.T, eye))
            for b, gb in enumerate(space.groups):
                c = C[ga][:, gb]
                if c.nnz:
                    blocks[a][b] += sp.kron(c.conjugate(), c)
    L = clean(sp.bmat(blocks, format='csr'))
    if maxabs(space.trace_row @ L) > 1e-11:
        raise ValueError('Reduced generator is not trace preserving')
    return L


def dissipator_action(C, rho):
    C = sp.csr_matrix(C); m = C.getH() @ C
    right = lambda a, b: (b.T @ a.T).T
    return right(C @ rho, C.getH()) - .5*(m @ rho + right(rho, m))

def maxabs(a):
    values = a.data if sp.issparse(a) else np.asarray(a)
    return float(np.max(np.abs(values), initial=0.0))

def clean(a):
    a = a.tocsr()
    a.sum_duplicates()
    a.eliminate_zeros()
    a.sort_indices()
    return a

def ketbra(d, a, b):
    return sp.csr_matrix(([1.0 + 0j], ([a], [b])), shape=(d, d))

def tensor(factors):
    out = sp.csr_matrix([[1.0 + 0j]])
    for a in factors:
        out = sp.kron(out, a, format='csr')
    return out

def build_model(p):
    if 'rr_rates' in p:
        if p['g'] != 0:

        if set(p['rr_rates']) != {'01','12','02'}:

                raise ValueError('Invalid RR rates')
    if p['gate_form'] not in {'Pi', 'Q'} or p['gate_idle'] not in {'Pi', 'Q', 'S'}:
        raise ValueError('Unknown gate')
    dims = [2, 2, 2, 3, 3, 3]

    size = int(np.prod(dims))
    eye = sp.eye(size, format='csr', dtype=complex)

    def embed(a, at):
        factors = identities.copy()
        factors[at] = a
        return tensor(factors)

    plus = [embed(ketbra(2, 1, 0), i) for i in range(3)]

    for e, (i, j) in enumerate(edge_pairs):

    return clean(h_hop + h_form), jumps, clean(sum(prs) + sum(pls)), clean(p['eta'] * sum(p0s) - p['delta'] * sum(pls)), dims

def build_liouvillian(h, jumps):
    eye = sp.eye(h.shape[0], format='csr', dtype=complex)
    result = -1j * (sp.kron(eye, h, format='csr') - sp.kron(h.T, eye, format='csr'))
    for _, c in jumps:
        a = clean(c.getH() @ c)
        result += sp.kron(c.conjugate(), c, format='csr')
        result -= 0.5 * sp.kron(eye, a, format='csr')
        result -= 0.5 * sp.kron(a.T, eye, format='csr')
    return clean(result)

def right(rho, a):
    return (a.T @ rho.T).T

def make_direct_rhs(h, jumps):
    products = [(c, c.getH().tocsr(), clean(c.getH() @ c)) for _, c in jumps]

    def rhs(v):
        rho = v.reshape(h.shape, order='F')
        out = -1j * (h @ rho - right(rho, h))
        for c, ch, a in products:
            out += right(c @ rho, ch) - 0.5 * (a @ rho + right(rho, a))
        return out.reshape(-1, order='F')
    return rhs

def rk4(rhs, v, dt):
    a = rhs(v)
    b = rhs(v + dt * a / 2)
    c = rhs(v + dt * b / 2)
    d = rhs(v + dt * c)
    return v + dt * (a + 2*b + 2*c + d) / 6

def adjoint_jump(c, observable):
    a = c.getH() @ c
    return clean(c.getH() @ observable @ c - (a @ observable + observable @ a) / 2)

def null_projector(a):
    _, s, vh = np.linalg.svd(a, full_matrices=True)
    tol = 1e-10*max(1., s[0] if len(s) else 0.)
    v = vh[int(np.sum(s > tol)):].conj().T
    p = v@v.conj().T
    return p, dict(singular_values=s.tolist(), tolerance=tol,
                   dimension=v.shape[1], residual=float(np.linalg.norm(a@p)))

def dark_projectors(h):
    c = np.array([[0., 0., 1.]])
    k, ki = null_projector(np.vstack([c, c@h]))
    d, di = null_projector(np.vstack([c, c@h, c@h@h]))
    connected = {2}
    for _ in range(3):
        connected |= {j for i in list(connected) for j in range(3)
                      if i != j and abs(h[i, j]) > 1e-12}
    isolated = np.diag([float(i not in connected) for i in range(3)])
    # Every sink-disconnected component is invariant; separate it from interference.
    return k, d, isolated, d-isolated, dict(K=ki, D=di)
