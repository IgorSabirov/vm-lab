import numpy as np
from scipy import sparse as sp
from scipy.sparse.linalg import splu

def renewal_check(model, state, tolerance=1e-9):
    if model.get('original_dimension') != 4:
        return dict(status='NOT_APPLICABLE')
    space = model['space']; groups = [i for i, g in enumerate(space.groups) if space.charges[g[0]] == 1]
    if len(groups) != 1:
        return dict(status='UNRESOLVED_EXCITATION_BLOCK')
    group = groups[0]; lo, hi = space.offsets[group:group+2]
    # Injection state is generated by the physical source from the empty state.
    C = dict(model['jumps'])['source_0']; empty = np.flatnonzero(space.charges == 0)
    if len(empty) != 1:
        return dict(status='UNRESOLVED_EMPTY_BRANCH')
    rho0 = np.zeros((space.d, space.d), complex); rho0[empty[0], empty[0]] = 1.
    injection = C@rho0@C.getH(); P = float(np.trace(injection).real)
    b = space.pack(injection/P)[lo:hi]; K = model['L'][lo:hi, lo:hi]
    try:
        R = splu(K.tocsc()).solve(-b)
    except RuntimeError:
        return dict(status='DARK_OR_SINGULAR_BLOCK_REQUIRES_BRANCH_ANALYSIS')
    full_R = np.zeros(space.size, complex); full_R[lo:hi] = R
    tau = float(space.trace(full_R).real)
    if tau <= 0 or maxabs(K@R+b) > tolerance:
        return dict(status='INVALID_RENEWAL_SOLUTION')
    predicted = P*full_R/(1+P*tau)
    ground = np.zeros((space.d, space.d), complex); ground[empty[0], empty[0]] = 1/(1+P*tau)
    predicted += space.pack(ground)
    error = maxabs(predicted-state)
    return dict(status='PASS' if error <= tolerance else 'FAIL', tau=tau, P_tau=P*tau,
                predicted_F=P/(1+P*tau), predicted_F_over_p_e=1/tau, state_error_inf=error)


def symmetry_check(model, tolerance=1e-12):
    """Test W K relating epsilon and -epsilon at Phi=pi/2, before using parity."""
    if model['space'].d != 216 or abs(model['case']['phi_over_pi']-.5) > 1e-12:
        return dict(symmetry_applicable=False, operator_covariance_residual=None,
                    parity_residual_max=None, parity_tolerance=None, parity_status='NOT_APPLICABLE')
    edges = np.array(list(np.ndindex(3, 3, 3)))
    diagonal = np.array([(-1.)**(((node >> 1) & 1)+sum(edge == 2)) for node in range(8) for edge in edges])
    W = sp.diags(diagonal); H = model['H']; epsilon = model['case']['epsilon']
    err = maxabs((H-2*epsilon*model['n'][1])+W@H.conjugate()@W)
    for _, C in model['jumps']:
        transformed = W@C.conjugate()@W
        # All declared jumps transform up to a real sign in this convention.
        err = max(err, min(maxabs(transformed-C), maxabs(transformed+C)))
    P = model['observables']['P_m6_N2']; J = model['observables']['h6']
    err = max(err, maxabs(W@P.conjugate()@W-P), maxabs(W@J.conjugate()@W-J))
    return dict(symmetry_applicable=err <= tolerance, operator_covariance_residual=err,
                parity_residual_max=None, parity_tolerance=None,
                parity_status='OPERATOR_CHECK_PASSED_STATE_PAIRS_PENDING' if err <= tolerance else 'OPERATOR_CHECK_FAILED')


def f2_differences(records, baseline):
    # Baseline is a pinned, previously verified report of undeformed cases.
    from .plan import full_physical_key
    refs = {}
    for r in baseline['records']:
        if r['case'].get('m6_v_over_J') != 1.:
            raise ValueError('Baseline must contain only undeformed cases')
        key = full_physical_key(r['case'])
        if r.get('physical_key') != key:
            raise ValueError('Baseline lacks matching complete physical fingerprint')
        if key in refs:
            raise ValueError('Duplicate baseline physical case')
        refs[key] = r
    if baseline.get('status') != 'VERIFIED_GITHUB':
        raise ValueError('Unverified baseline')
    out = []
    for record in records:
        c = record['case']; key = full_physical_key(c, undeformed=True)
        if key not in refs:
            raise ValueError('F2 lacks matching baseline')
        base = refs[key]; changes = {name: record['raw'][name]-base['raw'][name] for name in ('J_h', 'h6', 'P_m6_N2')}
        for name in ('J_h', 'h6', 'P_m6_N2'):
            a_u, b_u = record.get('primitive_uncertainty', {}).get(name), base.get('primitive_uncertainty', {}).get(name)
            changes[name+'_uncertainty'] = None if a_u is None or b_u is None else a_u+b_u
        a, b = record['conditional']['j6'], base['conditional']['j6']
        changes['j6'] = None if a['value'] is None or b['value'] is None else a['value']-b['value']
        changes['j6_uncertainty'] = None if a['uncertainty'] is None or b['uncertainty'] is None else a['uncertainty']+b['uncertainty']
        out.append(dict(id=record['id'], baseline_id=base['id'], changes=changes))
    return out


def measured_symmetries(records, responses):
    groups = {}; results = []
    for r in records:
        c = r['case']
        if not r.get('symmetry', {}).get('symmetry_applicable'):
            continue
        key = (c['model'], c['phi_over_pi'], c['node_noise'], c['m6_v_over_J'])
        groups.setdefault(key, {})[float(c['epsilon'])] = r
    for key, line in groups.items():
        for eps in sorted(e for e in line if e > 0 and -e in line):
            a, b = line[eps], line[-eps]
            for name in ('J_h', 'h6', 'P_m6_N2'):
                error = abs(a['raw'][name]-b['raw'][name])
                limit = a['primitive_uncertainty'][name]+b['primitive_uncertainty'][name]
                results.append(dict(kind='antiunitary_parity', cases=[a['id'], b['id']], observable=name,
                                    error=float(error), tolerance=float(limit), status='PASS' if error <= limit else 'FAIL'))
    freq = {}
    for r in responses:
        if r['kind'] == 'frequency':
            freq.setdefault(r['base_id'], {})[r['omega']] = r
    for case, line in freq.items():
        if -.5 not in line or .5 not in line:
            continue
        a, b = line[.5], line[-.5]
        for name in ('J_h', 'K', 'P_m6_N2', 'j6'):
            if a['values'].get(name) is None or b['values'].get(name) is None:
                continue
            error = abs(a['values'][name]-np.conj(b['values'][name]))
            limit = a['uncertainty'][name]+b['uncertainty'][name]
            results.append(dict(kind='frequency_conjugacy', cases=[case], observable=name,
                                error=float(error), tolerance=float(limit), status='PASS' if error <= limit else 'FAIL'))
    return results


def evaluate_predictions(predictions, records, differences):
    """Numerical verdict only when the requested metric and error are resolved."""
    records = {r['id']: r for r in records}; diffs = {r['id']: r for r in differences}; out = []
    for p in predictions['predictions']:
        result = dict(prediction_id=p['prediction_id'], status='EXPLORATORY')
        if p['confirmatory_or_exploratory'] == 'exploratory':
            out.append(result); continue
        definition = p['observable_definition']; name = definition['observable']; terms = definition['linear_terms']
        value = 0.; uncertainty = 0.; unresolved = False; missing_error = False
        for term in terms:
            case_id = term['case_id']; coefficient = float(term['coefficient']); r = records[case_id]
            if definition['kind'] == 'baseline_difference':
                d = diffs[case_id]['changes']; v = d.get(name); u = d.get(name+'_uncertainty')
            elif name in r['conditional']:
                v, u = r['conditional'][name]['value'], r['conditional'][name]['uncertainty']
            else:
                v = r['raw'].get(name); u = r.get('primitive_uncertainty', {}).get(name)
            if v is None:
                unresolved = True; continue
            value += coefficient*v
            if u is None:
                missing_error = True
            else:
                uncertainty += abs(coefficient)*u
        result.update(value=None if unresolved else float(value), uncertainty=None if missing_error else float(uncertainty))
        if unresolved or missing_error:
            result['status'] = 'NUMERICALLY_UNRESOLVED'; out.append(result); continue
        tol = p['tolerance_and_error_rule']['absolute_tolerance']; expected = p['expected_value_or_sign_or_interval']
        if p['prediction_type'] == 'value':
            lower, upper = float(expected)-tol, float(expected)+tol
        elif p['prediction_type'] == 'interval':
            lower, upper = float(expected[0])-tol, float(expected[1])+tol
        else:
            lower, upper = (tol, np.inf) if expected == 1 else (-np.inf, -tol)
        a, b = value-uncertainty, value+uncertainty
        result['status'] = 'SUPPORTED_WITHIN_DECLARED_TOLERANCE' if a >= lower and b <= upper else ('PREDICTION_FAILED' if b < lower or a > upper else 'NUMERICALLY_UNRESOLVED')
        out.append(result)
    return out

def serial(value):
    if isinstance(value, complex):
        if not np.isfinite([value.real, value.imag]).all():
            raise ValueError('Nonfinite complex output')
        return dict(re=float(value.real), im=float(value.imag))
    if isinstance(value, np.ndarray):
        return serial(value.tolist())
    if isinstance(value, np.generic):
        return serial(value.item())
    if isinstance(value, dict):
        return {k: serial(v) for k, v in value.items()}
    if isinstance(value, (tuple, list)):
        return [serial(v) for v in value]
    return value


def real_values(values, tolerance=1e-9):
    out = {}
    for name, v in values.items():
        if name.startswith('C_'):
            out[name] = complex(v)
        elif abs(complex(v).imag) > tolerance:
            raise ArithmeticError('Hermitian observable has imaginary residue: '+name)
        else:
            out[name] = float(complex(v).real)
    return out


def conditional_values(raw, errors=None):
    errors = errors or {}
    def conditional(h, p):
        return ratio(raw[h], raw[p], errors.get(h), errors.get(p))
    out = {}
    for key in raw:
        if key.startswith('p_m') and '_N' in key:
            suffix = key[1:]; out['j'+suffix] = conditional('h'+suffix, key)
        if key.startswith('p_N'):
            out['j'+key[1:]] = conditional('h'+key[1:], key)
    if 'P_m6_N2' in raw:
        out['j6'] = conditional('h6', 'P_m6_N2')
    if 'p_e' in raw:
        out['F_over_p_e'] = conditional('F', 'p_e')
    return out


def validate_conditionals(raw, conditioned, errors=None):
    expected = conditional_values(raw, errors)
    if set(expected) != set(conditioned):
        raise ValueError('Conditional fields mismatch')
    for key, item in expected.items():
        actual = conditioned[key]['value']
        if (item['value'] is None) != (actual is None):
            raise ValueError('Conditional null policy mismatch')
        if actual is not None and (not np.isfinite(actual) or abs(actual-item['value']) > 1e-12*max(1., abs(actual))):
            raise ValueError('Conditional reconstruction mismatch')
        if conditioned[key].get('uncertainty') != item['uncertainty']:
            raise ValueError('Conditional uncertainty mismatch')


def summaries(records):

        return tensor([x if k == at else sp.eye(2, format='csr') for k in range(3)])

"""Bordered sparse solves; state and response checks deliberately differ."""
import time
import numpy as np
from scipy import sparse as sp
from scipy.sparse.linalg import splu, spilu, gmres, LinearOperator, onenormest
from e002_exp.model import maxabs

FLOOR = 1e-6


def refinement_check(primary, reference, values_primary, values_reference, limits):
    """Calibrated engineering acceptance thresholds; not a rigorous error bound."""
    state_difference = maxabs(np.asarray(primary)-np.asarray(reference))
    errors = {name: float(abs(value-values_reference[name])) for name, value in values_primary.items()
              if value is not None and values_reference.get(name) is not None}
    failed = [name for name, error in errors.items() if error > limits['observable_absolute']+
              limits['observable_relative']*max(abs(values_primary[name]), abs(values_reference[name]))]
    passed = state_difference <= limits['vector_inf'] and not failed
    return dict(pass_gate=bool(passed), vector_difference_inf=state_difference, observable_differences=errors,
                failed_observables=failed, limits=limits, status='PASS' if passed else 'NUMERICALLY_UNRESOLVED')


def residual(A, x, b):
    r = A@x-b
    return dict(absolute_inf=maxabs(r), relative_2=float(np.linalg.norm(r)/max(np.linalg.norm(b), 1.)))


class Bordered:
    def __init__(self, L, space, z=0j, method='lu', rtol=1e-11, atol=1e-13,
                 maxiter=400, restart=100, ilu_drop_tol=1e-5, ilu_fill_factor=10.):
        self.space = space; self.L = sp.csr_matrix(L); self.z = complex(z)
        self.A = self.L if z == 0 else (z*sp.eye(space.size, format='csr')-self.L)
        r = space.trace_row.T/space.d
        self.matrix = sp.bmat([[self.A, r], [space.trace_row, None]], format='csc')
        self.rtol, self.atol, self.maxiter, self.restart = rtol, atol, maxiter, restart
        self.method = method; self.factor = None; self.preconditioner = None
        self.iterations = 0; self.solves = 0; self.solve_seconds = 0.
        start = time.perf_counter()
        if method == 'lu':
            self.factor = splu(self.matrix)
        elif method == 'gmres_ilu':
            self.factor = spilu(self.matrix, drop_tol=ilu_drop_tol, fill_factor=ilu_fill_factor)
            self.preconditioner = LinearOperator(self.matrix.shape, matvec=self.factor.solve, dtype=complex)
        else:
            raise ValueError('Unsupported solver')
        self.factor_seconds = time.perf_counter()-start
        self.factor_nnz = int(self.factor.L.nnz+self.factor.U.nnz)
        self.factor_storage_bytes = int(sum(a.data.nbytes+a.indices.nbytes+a.indptr.nbytes for a in (self.factor.L, self.factor.U)))

    def solve(self, b=None, trace=0.):
        b = np.zeros(self.space.size, complex) if b is None else np.asarray(b, complex)
        if b.shape != (self.space.size,) or not np.isfinite(b).all():
            raise ValueError('Invalid RHS')
        if abs(self.space.trace(b)) > max(1e-11, 10*self.atol):
            raise ValueError('RHS must have zero trace')
        if self.z != 0 and trace != 0:
            raise ValueError('Nonzero frequency expects traceless response')
        rhs = np.r_[b, trace]; start = time.perf_counter(); iterations = 0
        if self.method == 'lu':
            x = self.factor.solve(rhs)
            # Residual correction, never a state projection or renormalization.
            for _ in range(2):
                correction = self.factor.solve(rhs-self.matrix@x)
                x += correction
        else:
            def count(_):
                nonlocal iterations
                iterations += 1
            x, info = gmres(self.matrix, rhs, M=self.preconditioner, rtol=self.rtol,
                            atol=self.atol, maxiter=self.maxiter, restart=self.restart,
                            callback=count, callback_type='pr_norm')
            if info != 0:
                raise ArithmeticError('GMRES did not converge: '+str(info))
        self.solve_seconds += time.perf_counter()-start; self.solves += 1; self.iterations += iterations
        checks = residual(self.A, x[:-1], b)
        checks.update(trace_error=float(abs(self.space.trace(x[:-1])-trace)), border_multiplier=float(abs(x[-1])), iterations=iterations)
        limit = max(10*self.atol, 10*self.rtol*max(1., np.linalg.norm(b)))
        if not np.isfinite(x).all() or max(checks['absolute_inf'], checks['trace_error'], checks['border_multiplier']) > limit:
            raise ArithmeticError('Original equation/trace residual failed: '+str(checks))
        return x[:-1], checks

    def condition_estimate(self):
        if self.method != 'lu':
            return dict(value=None, status='NOT_ESTIMATED_FOR_ITERATIVE_SOLVER')
        inverse = LinearOperator(self.matrix.shape, matvec=self.factor.solve,
                                 rmatvec=lambda x: self.factor.solve(x, trans='H'), dtype=complex)
        estimate = float(sp.linalg.norm(self.matrix, 1)*onenormest(inverse))
        return dict(value=estimate, status='ONE_NORM_ESTIMATE_NOT_RIGOROUS_ERROR_BOUND')

    def metrics(self):
        return dict(method=self.method, matrix_nnz=int(self.matrix.nnz), factor_nnz=self.factor_nnz,
                    factor_storage_bytes=self.factor_storage_bytes, factor_seconds=self.factor_seconds,
                    solves=self.solves, solve_seconds=self.solve_seconds, iterations=self.iterations)


def state_checks(L, space, x, tolerance=1e-10):
    rho = space.unpack(x)
    eigmin = min(float(np.linalg.eigvalsh((rho[np.ix_(g, g)]+rho[np.ix_(g, g)].conj().T)/2).min()) for g in space.groups)
    result = dict(trace_error=float(abs(space.trace(x)-1)), hermiticity=maxabs(rho-rho.conj().T),
                  minimum_eigenvalue=eigmin, Lrho_inf=maxabs(L@x), Lrho_2=float(np.linalg.norm(L@x)))
    result['pass'] = bool(result['trace_error'] <= tolerance and result['hermiticity'] <= tolerance
                          and eigmin >= -tolerance and result['Lrho_inf'] <= tolerance)
    return result

def ratio(h, p, uh=None, up=None):
    if not np.isfinite([h, p]).all() or p <= FLOOR:
        return dict(value=None, uncertainty=None, status='SMALL_DENOMINATOR')
    j = h/p
    if uh is None or up is None:
        return dict(value=float(j), uncertainty=None, status='ERROR_UNRESOLVED')
    if min(uh, up) < 0 or not np.isfinite([uh, up]).all():
        raise ValueError('Invalid error estimates')
    if p-up <= FLOOR:
        return dict(value=None, uncertainty=None, status='DENOMINATOR_UNRESOLVED')
    return dict(value=float(j), uncertainty=float((uh+abs(j)*up)/(p-up)), status='PROPAGATED_ESTIMATE')


def contrast(left, center, right, delta):
    if delta <= 0:
        raise ValueError('delta must be positive')
    if any(x['value'] is None for x in (left, center, right)):
        return dict(D=None, d=None, curvature=None, uncertainty_D=None, status='UNDEFINED')
    D = center['value']-(left['value']+right['value'])/2
    u = None if any(x['uncertainty'] is None for x in (left, center, right)) else center['uncertainty']+(left['uncertainty']+right['uncertainty'])/2
    return dict(D=float(D), d=None if abs(center['value']) <= FLOOR else float(-D/center['value']),
                curvature=float(-2*D/delta**2), uncertainty_D=u,
                center_below_both=bool(center['value'] < min(left['value'], right['value'])),
                status='ERROR_UNRESOLVED' if u is None else 'PROPAGATED_ESTIMATE')

def qrt_inputs(model, state, names=('J_h', 'P_m6_N2', 'K')):
    space = model['space']; rho = space.unpack(state); base = values(model, state)
    inputs = []; variances = []
    for name in names:
        B = model['observables'][name]; mean = base[name]
        if abs(mean.imag) > 1e-10:
            raise ArithmeticError('Hermitian mean is complex')
        y = space.pack(B@rho-mean*rho)
        if abs(space.trace(y)) > 1e-10:
            raise ArithmeticError('QRT input trace failed')

def propagate_qrt(model, state, horizon, dt, sample_interval, names=('J_h', 'P_m6_N2', 'K'), checkpoint=None):
    if min(horizon, dt, sample_interval) <= 0:
        raise ValueError('Invalid QRT grid')
    steps = round(horizon/dt); stride = round(sample_interval/dt)
    if stride < 1 or abs(steps*dt-horizon) > 1e-9 or abs(stride*dt-sample_interval) > 1e-9 or steps % stride:
        raise ValueError('QRT grid must have integral steps and common final sample')
    y, var = qrt_inputs(model, state, names); L = model['L']; space = model['space']
    rows = np.vstack([space.row(model['observables'][name]) for name in names])
    times = [0.]; out = [rows@y]; trace_max = maxabs(space.trace(y)); start = time.perf_counter()
    for step in range(1, steps+1):
        a = L@y; b = L@(y+dt*a/2); c = L@(y+dt*b/2); d = L@(y+dt*c)
        y += dt*(a+2*b+2*c+d)/6
        if step % stride == 0:
            if not np.isfinite(y).all():
                raise ArithmeticError('QRT became nonfinite')
            t = step*dt; times.append(t); out.append(rows@y)
            trace_max = max(trace_max, maxabs(space.trace(y)))
            if checkpoint:
                checkpoint(step, t, y)
    if trace_max > 1e-8:
        raise ArithmeticError('QRT trace drift: '+str(trace_max))
    times = np.array(times); out = np.array(out); normalized = np.full_like(out, np.nan)
    den = np.sqrt(var[:, None]*var[None, :]); valid = den > FLOOR
    normalized[:, valid] = out[:, valid]/den[valid]
    integral = np.trapezoid(abs(out.diagonal(axis1=1, axis2=2)), times, axis=0)
    tau = [float(a/v) if v > FLOOR else None for a, v in zip(integral, var)]
    late = out[int(.8*len(out)):]
    summary = dict(steps=steps, block_rhs_calls=4*steps, scalar_rhs_equivalents=4*steps*len(names),
                   inputs=len(names), outputs_per_input=len(names), samples=len(times), trace_max=trace_max,
                   elapsed_seconds=time.perf_counter()-start, variances=var.tolist(), tau_int_truncated=tau,
                   late_window_abs_max=float(abs(late).max()), tail_status='TAIL_UNRESOLVED',
                   infinite_integral=None, input_convention='(B-mean_B)rho; ordered connected QRT',
                   positivity_test_applied=False)
    return dict(times=times, C=out, normalized=normalized, normalization_valid=valid, summary=summary, final=y)

def schur_response(L, b, row, target, z, rtol=1e-10, maxiter=300):
    """Complete T/B partition; matrix-free Schur complement, no dense inverse.

    This candidate decomposition alone cannot certify Fano. z=0 is deliberately
    rejected here: stationary constraints need the separately validated border.
    """
    if z == 0:
        raise ValueError('Schur candidate at z=0 requires a separately declared constrained partition')
    L = sp.csr_matrix(L); n = L.shape[0]; target = np.asarray(target, dtype=int)
    if target.ndim != 1 or len(np.unique(target)) != len(target) or not 0 < len(target) < n or min(target) < 0 or max(target) >= n:
        raise ValueError('Invalid Schur partition')
    other = np.setdiff1d(np.arange(n), target); A = z*sp.eye(n, format='csr')-L
    tt = A[target][:, target]; tb = A[target][:, other]; bt = A[other][:, target]; bb = A[other][:, other]
    factor = splu(bb.tocsc()); rb_b = factor.solve(np.asarray(b)[other])
    eff = LinearOperator(tt.shape, matvec=lambda x: tt@x-tb@factor.solve(bt@x), dtype=complex)
    rhs = np.asarray(b)[target]-tb@rb_b
    xt, info = gmres(eff, rhs, rtol=rtol, atol=rtol*1e-2, maxiter=maxiter)
    if info:
        raise ArithmeticError('Schur GMRES did not converge')
    xb = factor.solve(np.asarray(b)[other]-bt@xt)
    x = np.empty(n, complex); x[target] = xt; x[other] = xb
    err = maxabs(A@x-b)
    if err > 10*rtol*max(1., np.linalg.norm(b)):
        raise ArithmeticError('Schur reconstruction residual failed')
    row = np.asarray(row); bg = complex(row[other]@rb_b)
    mediated = complex(row[target]@xt-row[other]@factor.solve(bt@xt))
    total = complex(row@x)
    return dict(x=x, background=bg, mediated=mediated, total=total,
                cross_intensity=float(2*np.real(bg.conjugate()*mediated)), reconstruction_error=float(abs(total-bg-mediated)),
                equation_residual=err, target_size=len(target), complement_size=len(other),
                interpretation='CANDIDATE_DECOMPOSITION_REQUIRES_PHYSICAL_PARTITION_AND_HELDOUT_CHECKS')


def g0(epsilon, v, J=1., kappa_left=.5, kappa_right=.5):
    A = J*J+kappa_left*kappa_right/4; width = kappa_left*v*v/(2*A)
    bg = -J*np.sqrt(kappa_left*kappa_right)/A
    if v == 0:
        amplitude, resonance = complex(bg), 0j
    else:
        amplitude = bg*epsilon/(epsilon-1j*width); resonance = amplitude-bg
    return dict(amplitude=amplitude, background=complex(bg), resonance=resonance,
                intensity=float(abs(amplitude)**2), width=width,
                cross=float(2*np.real(complex(bg).conjugate()*resonance)))

"""Charge-covariant q=0 Liouvillian, without a dense Liouville matrix."""
from dataclasses import dataclass
import numpy as np
from scipy import sparse as sp
from e002_exp.model import clean, maxabs


@dataclass
class Space:
    charges: np.ndarray

    def __post_init__(self):
        self.charges = np.asarray(self.charges, dtype=int)
        self.d = len(self.charges)
        self.groups = [np.flatnonzero(self.charges == n) for n in np.unique(self.charges)]
        self.sizes = [len(g) ** 2 for g in self.groups]
        self.offsets = np.r_[0, np.cumsum(self.sizes)]
        self.indices = np.concatenate([(g[:, None] + self.d*g[None, :]).ravel(order='F') for g in self.groups])
        self.trace_indices = np.concatenate([off + np.arange(len(g))*(len(g)+1)
                                            for off, g in zip(self.offsets, self.groups)])
        self.size = len(self.indices)
        self.trace_row = sp.csr_matrix((np.ones(self.d), (np.zeros(self.d, int), self.trace_indices)), shape=(1, self.size))

    def pack(self, matrix, tolerance=1e-12):
        a = matrix.toarray() if sp.issparse(matrix) else np.asarray(matrix)
        if a.shape != (self.d, self.d):
            raise ValueError('Hilbert dimensions disagree')
        if maxabs(np.where(self.charges[:, None] != self.charges[None, :], a, 0)) > tolerance:
            raise ValueError('Input has components outside q=0')
        return a.ravel(order='F')[self.indices].copy()

    def unpack(self, vector):
        v = np.asarray(vector)
        if v.shape != (self.size,):
            raise ValueError('Reduced vector shape mismatch')
        a = np.zeros(self.d*self.d, complex)
        a[self.indices] = v
        return a.reshape(self.d, self.d, order='F')

    def row(self, operator):
        # trace(O rho), without conjugation of O: vec(O.T)^T vec(rho).
        a = operator.toarray() if sp.issparse(operator) else np.asarray(operator)
        return a.T.ravel(order='F')[self.indices]

    def trace(self, vector):
        return np.asarray(vector)[self.trace_indices].sum(axis=0)


def charge_checks(H, jumps, space, tolerance=1e-12):
    h = sp.coo_matrix(H)
    hbad = maxabs(h.data[space.charges[h.row] != space.charges[h.col]])
    deltas = {}
    for name, C in jumps:
        c = sp.coo_matrix(C)
        nz = abs(c.data) > tolerance
        ds = np.unique(space.charges[c.row[nz]] - space.charges[c.col[nz]])
        if len(ds) > 1:
            raise ValueError('Jump lacks definite charge: ' + name)
        deltas[name] = int(ds[0]) if len(ds) else None
    if hbad > tolerance:
        raise ValueError('Hamiltonian violates charge covariance')
    return dict(H_charge_residual=hbad, jump_charge_changes=deltas)


def liouvillian(H, jumps, space):
    """Exact block construction; every within-charge coherence is retained."""
    charge_checks(H, jumps, space)
    H = sp.csr_matrix(H)
    count = len(space.groups)
    blocks = [[sp.csr_matrix((space.sizes[a], space.sizes[b]), dtype=complex)
               for b in range(count)] for a in range(count)]
    for a, g in enumerate(space.groups):
        h = H[g][:, g]; eye = sp.eye(len(g), format='csr')
        blocks[a][a] += -1j*(sp.kron(eye, h) - sp.kron(h.T, eye))
    for _, C in jumps:
        C = sp.csr_matrix(C); loss = clean(C.getH() @ C)
        for a, ga in enumerate(space.groups):
            eye = sp.eye(len(ga), format='csr'); m = loss[ga][:, ga]
            blocks[a][a] -= .5*(sp.kron(eye, m) + sp.kron(m.T, eye))
            for b, gb in enumerate(space.groups):
                c = C[ga][:, gb]
                if c.nnz:
                    blocks[a][b] += sp.kron(c.conjugate(), c)
    L = clean(sp.bmat(blocks, format='csr'))
    if maxabs(space.trace_row @ L) > 1e-11:
        raise ValueError('Reduced generator is not trace preserving')
    return L


def dissipator_action(C, rho):
    C = sp.csr_matrix(C); m = C.getH() @ C
    right = lambda a, b: (b.T @ a.T).T
    return right(C @ rho, C.getH()) - .5*(m @ rho + right(rho, m))

from .observations import static_response, frequency_response, propagate_qrt, compare_qrt
from .plan import preflight, workload, pinned_file, full_physical_key
from .analysis import renewal_check, symmetry_check, f2_differences, evaluate_predictions, measured_symmetries
from .runtime import environment, runtime_gate, kernel_hash
from .export import serial, real_values, conditional_values, summaries, save_case, save_correlations, write_manifest, write_result, verify_report


def _run(config, output=Path('.')):
    started = time.perf_counter(); output = Path(output)
    report = output/'report'; full = output/'full'
    report.mkdir(parents=True, exist_ok=False); full.mkdir(parents=True, exist_ok=False)
    env_before = environment(); work, env_check = preflight(config, env_before)
    save_json(report/'config.json', config); save_json(report/'environment_before.json', env_before)
    records = []; responses = []; qrts = []; phases = []; failure = None
    actual = dict.fromkeys(config['expected_workload'], 0)
    solver_options = config['numerics']['solver']; quality_tolerance = config['numerics']['state_tolerance']
    refinement_options = config['numerics']['refinement_solver']; error_policy = config['numerics']['error_envelope']
    def error_envelope(a, b):
        return {k: error_policy['absolute_floor']+error_policy['difference_multiplier']*float(abs(v-b[k]))
                for k, v in a.items() if v is not None and b.get(k) is not None}
    try:
        cases = [(f'small_{i:03d}', c, True) for i, c in enumerate(work['small'])]
        cases += [(c['id'], c, False) for c in work['full']]
        for case_id, case, small in cases:
            print('START', case_id, flush=True); start = time.perf_counter()
            model = make_small(case) if small else make_full(case)
            setup_time = time.perf_counter()-start; space = model['space']
            solver = Bordered(model['L'], space, **solver_options)
            state, solve_check = solver.solve(trace=1.)
            quality = state_checks(model['L'], space, state, quality_tolerance)
            if not quality['pass']:
                raise ArithmeticError('Unphysical or nonstationary NESS: '+str(quality))
            independent = Bordered(model['L'], space, **refinement_options)
            ref_state, ref_checks = independent.solve(trace=1.)
            ref_quality = state_checks(model['L'], space, ref_state, quality_tolerance)
            if not ref_quality['pass']:
                raise ArithmeticError('Independent NESS verification failed')
            actual['refinement_NESS'] += 1
            raw = real_values(values(model, state)); ref_raw = real_values(values(model, ref_state))
            refinement_gate = refinement_check(state, ref_state, raw, ref_raw, config['numerics']['refinement_limits'])
            if not refinement_gate['pass_gate']:
                save_json(report/(case_id+'_refinement_failure.json'), serial(refinement_gate))
                raise ArithmeticError('Independent NESS disagreement exceeds calibrated limits')
            errors = error_envelope(raw, ref_raw); conditional = conditional_values(raw, errors)
            if abs(raw['J_h']-raw['F']) > quality_tolerance*10 or abs(raw['source']-raw['F']) > quality_tolerance*10:
                raise ArithmeticError('Stationary flux balance failed')
            rec = dict(id=case_id, case=case, raw=raw, conditional=conditional, quality=quality,
                       physical_key=None if small else full_physical_key(case),
                       solve_check=solve_check, branch=model.get('branch', 'unique_NESS_candidate'),
                       operator_checks=model.get('checks', {}), hilbert_dimension=space.d,
                       liouville_dimension=space.size, generator_nnz=model['L'].nnz,
                       primitive_uncertainty=errors, refinement=dict(quality=ref_quality, solve=ref_checks,
                       gate=refinement_gate,
                       state_difference_inf=float(abs(state-ref_state).max()), error_policy=error_policy,
                       uncertainty_kind='ENGINEERING_TWO_METHOD_ENVELOPE_NOT_RIGOROUS_TOTAL_ERROR_BOUND'),
                       condition=solver.condition_estimate() if config['numerics'].get('estimate_condition', True) else {'status':'NOT_ESTIMATED'})
            rec['renewal'] = renewal_check(model, state) if small else {'status': 'NOT_APPLICABLE'}
            rec['symmetry'] = symmetry_check(model) if not small else {'symmetry_applicable': False, 'parity_status': 'SMALL_GAUGE_TESTED_SEPARATELY'}
            if rec['renewal']['status'] == 'FAIL':
                raise ArithmeticError('G1 renewal identity failed')
            if rec['symmetry'].get('parity_status') == 'OPERATOR_CHECK_FAILED':
                raise ArithmeticError('Declared antiunitary symmetry failed')
            save_case(report, full, rec, state, space); records.append(rec)
            actual['small_NESS' if small else 'full_NESS'] += 1
            for target in case.get('tasks', [])[1:]:
                x, out, checks = static_response(model, state, target, solver)
                xr, outr, checkr = static_response(model, ref_state, target, independent)
                agreement = refinement_check(x, xr, out, outr, config['numerics']['refinement_limits'])
                if not agreement['pass_gate']:
                    save_json(report/(case_id+'_'+target+'_refinement_failure.json'), serial(agreement))
                    raise ArithmeticError('Static response disagreement exceeds calibrated limits')
                responses.append(dict(base_id=case_id, kind='static', target=target, values=out, checks=checks,
                                      refinement_checks=checkr, refinement_gate=agreement, uncertainty=error_envelope(out, outr)))
                actual['static_responses'] += 1
                actual['refinement_static_responses'] += 1
            independent_metrics = independent.metrics(); del independent
            for freq in [f for f in work['frequency'] if f['base_id'] == case_id]:
                x, out, checks, met = frequency_response(model, state, freq['omega_over_J'], solver, **solver_options)
                xr, outr, checkr, metr = frequency_response(model, ref_state, freq['omega_over_J'], None, **refinement_options)
                agreement = refinement_check(x, xr, out, outr, config['numerics']['refinement_limits'])
                if not agreement['pass_gate']:
                    save_json(report/(case_id+'_frequency_refinement_failure.json'), serial(agreement))
                    raise ArithmeticError('Frequency response disagreement exceeds calibrated limits')
                responses.append(dict(base_id=case_id, kind='frequency', omega=freq['omega_over_J'], values=out,
                                      checks=checks, solver=met, refinement_checks=checkr, refinement_solver=metr,
                                      refinement_gate=agreement, uncertainty=error_envelope(out, outr)))
                actual['frequency_responses'] += 1
                actual['refinement_frequency_responses'] += 1
            corr = [c for c in work['correlations'] if c['base_id'] == case_id]
            if corr:
                names = tuple(c['input'] for c in corr); q = config['numerics']['qrt']
                def checkpoint(step, t, y):
                    # Checkpoint is explicit; automatic resume is intentionally not claimed.
                    if step % q['checkpoint_stride_steps'] == 0:
                        tmp = full/(case_id+'_qrt_checkpoint.tmp.npz')
                        np.savez_compressed(tmp, y=y, step=step, time=t, kernel_sha256=kernel_hash())
                        os.replace(tmp, full/(case_id+'_qrt_checkpoint.npz'))
                fine = propagate_qrt(model, state, q['horizon'], q['dt_fine'], q['sample_interval'], names, checkpoint)
                coarse = propagate_qrt(model, state, q['horizon'], q['dt_coarse'], q['sample_interval'], names)
                ref = compare_qrt(fine, coarse, q['absolute_tolerance'], q['relative_tolerance'])
                qrts.append(dict(id=case_id, **save_correlations(report, full, case_id, fine, coarse, ref)))
                actual['correlation_inputs'] += len(corr)
                if not ref['pass_gate']:
                    raise ArithmeticError('QRT refinement failed; finer continuation requires a declared budget')
            phases.append(dict(id=case_id, setup_seconds=setup_time, total_case_seconds=time.perf_counter()-start,
                               solver=solver.metrics(), refinement_solver=independent_metrics))
            save_json(report/'progress.json', dict(last_case=case_id, actual_counts=actual, elapsed_seconds=time.perf_counter()-started))
            print('DONE', case_id, flush=True)
            del solver, model, state; gc.collect()
    except Exception as exc:
        failure = type(exc).__name__+': '+str(exc)
        save_json(report/'failure.json', dict(message=failure, traceback=traceback.format_exc()))
    comparisons = summaries(records)
    symmetry_pairs = measured_symmetries(records, responses)
    save_json(report/'symmetry_pairs.json', serial(symmetry_pairs))
    if any(x['status'] == 'FAIL' for x in symmetry_pairs):
        failure = failure or 'Measured symmetry exceeds propagated numerical tolerance'
    if config['stage'] == 'F2' and not failure:
        differences = f2_differences(records, pinned_file(config['baseline_report']))
        verdicts = evaluate_predictions(pinned_file(config['predictions']), records, differences)
        save_json(report/'f2_differences.json', serial(differences))
        save_json(report/'prediction_verdicts.json', serial(verdicts))
    save_json(report/'contrasts.json', serial(comparisons)); save_json(report/'responses.json', serial(responses))
    elapsed = time.perf_counter()-started
    gate = runtime_gate(config['forecast'], elapsed, config['expected_workload'], actual,
                        env_check['comparable'], config['mode'] == 'production')
    if failure and gate['status'] == 'PASS':
        gate['status'] = 'REVIEW_REQUIRED'
    metrics = dict(runtime_gate=gate, phases=phases, environment_after=environment(),
                   environment_comparison=env_check, actual_counts=actual, qrt=qrts, failure=failure,
                   kernel_sha256=kernel_hash(), source_commit=os.environ.get('VM_SOURCE_COMMIT', config.get('source_commit')),
                   scientific_status='NOT_ACCEPTED_AUTOMATICALLY', full_process_time='USE_SERVICE_RUN_JSON')
    save_json(report/'metrics.json', serial(metrics))
    write_result(report, config, records, failure, gate, comparisons, qrts)
    from .export import write_full_manifest
    write_full_manifest(report, full)
    save_json(report/'status.json', dict(status='COMPLETED_NOT_ACCEPTED' if not failure else 'FAILED', failure=failure))
    write_manifest(report); verification = verify_report(report, full) if not failure else {'status': 'PARTIAL_FAILED_RUN'}
    print(json.dumps(dict(failure=failure, runtime_gate=gate, verification=verification)), flush=True)
    if failure:
        raise RuntimeError(failure)
    return metrics


def run(config, output=Path('.')):
    """Persist failure even in preflight/evaluation/export; never rewrite old runs."""
    output = Path(output); report = output/'report'
    if report.exists() or (output/'full').exists():
        raise FileExistsError('Output already contains a run; refusing to modify it')
    start = time.perf_counter()
    try:
        return _run(config, output)
    except Exception as exc:
        failure = dict(status='FAILED', type=type(exc).__name__, message=str(exc),
                       traceback=traceback.format_exc(), elapsed_seconds=time.perf_counter()-start,
                       scientific_acceptance=False)
        try:
            report.mkdir(parents=True, exist_ok=True)
            save_json(report/'terminal_failure.json', failure)
            save_json(report/'status.json', dict(status='FAILED', failure=failure['message']))
            metrics_path = report/'metrics.json'
            metrics = json.loads(metrics_path.read_text()) if metrics_path.exists() else {}
            metrics.update(failure=failure['message'], terminal_status='FAILED', scientific_status='NOT_ACCEPTED')
            gate = metrics.setdefault('runtime_gate', {})
            gate['status'] = 'REVIEW_REQUIRED' if gate.get('status') == 'PASS' else gate.get('status', 'NOT_EVALUABLE')
            save_json(metrics_path, metrics)
            (report/'RESULT.md').write_text('Время: NOT_EVALUABLE либо требует повторной проверки полного процесса.\n\n'
                'Расчёт завершён с ошибкой. Результат не принят.\n\n'+failure['type']+': '+failure['message']+'\n'
                'Подробности: terminal_failure.json. Частичные файлы сохраняются; прежний манифест не пересоздаётся для их принятия.\n')
        except Exception:
            print('FAILED_TO_PERSIST_TERMINAL_DIAGNOSTICS', traceback.format_exc(), flush=True)
        raise

import copy

def predictions():
    r = json.loads((ROOT/'experiments/e002e/f2_prereg_predictions.template.json').read_text())
    r.update(status='FROZEN', code_commit='a'*40, created_at_utc='2026-10-31T00:00:00Z', frozen_at_utc='2026-10-31T00:00:01Z',
             training_inputs=[dict(id='train', block='G', report_commit='b'*40, sha256='c'*64)])
    ids = r['heldout_case_ids']
    r['predictions'] = [dict(prediction_id='p1', model='QS/RR', physical_case_ids=ids,
                            observable_definition=dict(kind='linear_combination', observable='j6', linear_terms=[dict(case_id=i, coefficient=1./len(ids)) for i in ids]),
                            parameter_domain={'frozen': True}, prediction_type='value', expected_value_or_sign_or_interval=.3,
                            tolerance_and_error_rule=dict(absolute_tolerance=.01, error_rule='interval_with_numerical_uncertainty'),
                            training_input_ids=['train'], candidate_model_derivation='declared test candidate', denominator_validity_if_ratio='p>floor with propagated error',
                            outcome_on_failure='reject candidate', confirmatory_or_exploratory='confirmatory')]
    return r


class ReviewRegressions(unittest.TestCase):
    def test_finite_tolerance_and_external_cap(self):
        r = predictions(); validate_predictions(r, 'a'*40, {'p1': .02})
        for value in (float('inf'), float('nan'), 1e300, -.1, '0.01', True):
            bad = copy.deepcopy(r); bad['predictions'][0]['tolerance_and_error_rule']['absolute_tolerance'] = value
            with self.assertRaises(ValueError):
                validate_predictions(bad, 'a'*40, {'p1': .02})
        with self.assertRaises(ValueError):
            validate_predictions(r, 'a'*40)

    def test_bad_time_order_and_self_declared_training_rejected(self):
        r = predictions(); r['frozen_at_utc'] = '2026-10-30T00:00:00Z'
        with self.assertRaises(ValueError): validate_predictions(r, 'a'*40, {'p1': .02})
        r = predictions(); r['allowed_training_blocks'].append('SECRET_F2')
        r['training_inputs'][0]['block'] = 'SECRET_F2'
        with self.assertRaises(ValueError): validate_predictions(r, 'a'*40, {'p1': .02})

    def test_acceptable_residuals_do_not_override_refinement_disagreement(self):
        # Both states can lie in the kernel of a degenerate zero generator;
        # residual and trace alone cannot accept their distinct observable.
        x = np.array([.5, .5]); y = np.array([.6, .4])
        limits = dict(vector_inf=1e-6, observable_absolute=1e-6, observable_relative=1e-5)
        result = refinement_check(x, y, {'p': .5}, {'p': .6}, limits)
        self.assertFalse(result['pass_gate']); self.assertEqual(result['status'], 'NUMERICALLY_UNRESOLVED')

    def test_baseline_duplicates_overrides_and_error_propagation(self):
        b = dict(id='b', case=full_case(epsilon=0.), raw=dict(J_h=.2, h6=.1, P_m6_N2=.25),
                 conditional={'j6': {'value': .4, 'uncertainty': .002}},
                 primitive_uncertainty=dict(J_h=.001, h6=.001, P_m6_N2=.001))
        r = copy.deepcopy(b); r['id'] = 'r'; r['case']['m6_v_over_J'] = .5; r['raw']['J_h'] = .21
        b['physical_key'] = full_physical_key(b['case']); r['physical_key'] = full_physical_key(r['case'])
        baseline = dict(status='VERIFIED_GITHUB', records=[b])
        result = f2_differences([r], baseline)[0]
        self.assertAlmostEqual(result['changes']['J_h'], .01)
        self.assertAlmostEqual(result['changes']['J_h_uncertainty'], .002)
        bad = copy.deepcopy(baseline); bad['records'].append(b)
        with self.assertRaises(ValueError): f2_differences([r], bad)
        bad = copy.deepcopy(baseline); bad['records'][0]['case']['sink'] = 1.
        with self.assertRaises(ValueError): f2_differences([r], bad)

    def test_memory_budget_enforced(self):
        env = dict(memory_limit='1000', memory_current='100', memory_available='100 kB')
        budget = dict(peak_additional_bytes=500, reserve_bytes=100, measured_peak_rss_bytes=300, simultaneous_factor_storage_bytes=200)
        self.assertEqual(memory_preflight(budget, env)['status'], 'PASS')
        budget['peak_additional_bytes'] = 850
        with self.assertRaises(MemoryError): memory_preflight(budget, env)
        budget['measured_peak_rss_bytes'] = None
        with self.assertRaises(ValueError): memory_preflight(budget, env)

    def test_full_data_corruption_and_deletion_detected(self):
        for corrupt in (True, False):
            with tempfile.TemporaryDirectory() as temp:
                report = Path(temp)/'report'; full = Path(temp)/'full'; report.mkdir(); full.mkdir()
                p = full/'state.npz'; np.savez(p, x=np.array([1., 2.]))
                write_full_manifest(report, full); write_manifest(report)
                self.assertEqual(verify_report(report, full)['full_data'], 'VERIFIED')
                if corrupt: p.write_bytes(p.read_bytes()+b'bad')
                else: p.unlink()
                with self.assertRaises(ValueError): verify_report(report, full)

    def test_preflight_failure_is_persisted(self):
        work = dict(small=[small_case()], full=[], frequency=[], correlations=[])
        with tempfile.TemporaryDirectory() as temp, patch('e002e_exp.task.preflight', side_effect=ValueError('invalid provenance')):
            with self.assertRaises(ValueError): run(config(work, 'G'), Path(temp))
            status = json.loads((Path(temp)/'report/status.json').read_text())
            self.assertEqual(status['status'], 'FAILED')
            self.assertIn('invalid provenance', (Path(temp)/'report/terminal_failure.json').read_text())

    def test_final_export_failure_is_persisted_and_existing_run_untouched(self):
        work = dict(small=[small_case()], full=[], frequency=[], correlations=[])
        with tempfile.TemporaryDirectory() as temp, patch('e002e_exp.task.preflight', return_value=(work, {'comparable': True})), patch('e002e_exp.task.write_result', side_effect=ValueError('export failed')):

def config(work, stage):
    return dict(mode='production', stage=stage, source_commit='0'*40, expected_workload=workload(work),
                forecast=dict(expected_seconds=1., lower_seconds=.001, upper_seconds=30.),
                numerics=dict(solver=dict(method='lu', rtol=1e-12, atol=1e-14),
                              refinement_solver=dict(method='gmres_ilu', rtol=1e-12, atol=1e-14),

        model['observables'].update(P_m6_N2=model['observables']['p_e'], h6=model['observables']['J_h'])
