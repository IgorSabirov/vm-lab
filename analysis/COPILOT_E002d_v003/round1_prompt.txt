def analyze(results,mode):
    by={physics_key(r["summary"]["case"]):r for r in results};pairs={}
    for k,r in by.items():
        if r["summary"]["case"]["dt"]!=.005:continue
        other=(*k[:-1],.01)
        if other not in by:continue
        check=compare_pair(r,by[other]);pairs[r["summary"]["case"]["id"]]=check
        for obj in (r,by[other]):obj["summary"]["refinement_status"]="passed" if check["pass_gate"] else "failed"
    if mode!="production":
        return dict(pairs=pairs,A=[],B=[],C=[],scope="short technical run; no scientific interpretation")
    horizons=sorted({200.}|{float(r["summary"]["horizon"]) for r in results})
    cache={}
    def get(model,target="base",kind="base",gamma=0.,phi=0.,epsilon=0.,noise=.05,horizon=200.):
        key=(model,phi,epsilon,noise,target,kind,gamma,.005)
        ck=(key,horizon)
        if ck in cache:return cache[ck]
        if key not in by:return None
        fine=by[key];coarse=by.get((*key[:-1],.01))
        fm=integral_mean(fine["series"],fine["names"],horizon)
        cm=integral_mean(coarse["series"],coarse["names"],horizon) if coarse is not None else None
        if fm is None:return None
        times=horizon-10+2*np.arange(6)
        def ranges(r):
            idx=[int(np.argmin(abs(r["series"][:,0]-t))) for t in times]
            if not np.array_equal(r["series"][idx,0],times):raise ValueError("Exact window missing")
            return dict(zip(r["names"],np.ptp(r["series"][idx,1:],axis=0)))
        fr=ranges(fine);cr=ranges(coarse) if cm is not None else None
        vals={name:estimate(value,None if cm is None else
                CONTRACT["primitive_absolute_floor"]+CONTRACT["resolution_multiplier"]*

                       not r["summary"].get("failed") for r in allr)
        refined=matched and all(r["summary"].get("refinement_status")=="passed" for r in allr)
        reason="READY" if stationary and refined else "FINITE_TIME_ONLY" if not stationary else "SCREENING_ONLY" if not matched else "REFINEMENT_FAILED"

"""Explicit deterministic error envelopes; no statistical significance claims."""
import numpy as np
from .plan import CONTRACT
def estimate(value,uncertainty=None,reason=None):
    value=None if value is None else float(value)
    if value is None or not np.isfinite(value):
        return dict(value=None,lower=None,upper=None,uncertainty=None,reason=reason or "nonfinite")
    if uncertainty is None:
        return dict(value=value,lower=None,upper=None,uncertainty=None,reason=reason or "no_coarse_pair")
    u=float(uncertainty)
    if not np.isfinite(u) or u<0:raise ValueError("Invalid uncertainty")
    return dict(value=value,lower=value-u,upper=value+u,uncertainty=u,reason=reason)

def linear(terms):
    if any(x["value"] is None for _,x in terms):return estimate(None,reason="undefined_operand")
    value=sum(k*x["value"] for k,x in terms)
    if any(x["uncertainty"] is None for _,x in terms):return estimate(value)
    return estimate(value,sum(abs(k)*x["uncertainty"] for k,x in terms))

def ratio(a,b,probability=False):
    floor=CONTRACT["probability_floor"] if probability else CONTRACT["primitive_absolute_floor"]
    if a["value"] is None or b["value"] is None or abs(b["value"])<=floor or (probability and b["value"]<=floor):
        return estimate(None,reason="small_denominator")
    value=a["value"]/b["value"]
    if a["uncertainty"] is None or b["uncertainty"] is None:return estimate(value)
    lo,hi=b["lower"],b["upper"]
    if lo<=0<=hi or min(abs(lo),abs(hi))<=floor or b["uncertainty"]>.1*abs(b["value"]):
        return estimate(None,reason="uncertain_denominator")
    values=[x/y for x in (a["lower"],a["upper"]) for y in (lo,hi)]
    result=estimate(value,max(value-min(values),max(values)-value))
    # Retain the true interval; the symmetric radius is only a summary.
    result.update(lower=min(values),upper=max(values))
    return result

def signed(x,eligible=True):
    if not eligible:return "INELIGIBLE"
    if x["value"] is None:return "UNDEFINED"
    if x["uncertainty"] is None:return "SCREENING_ONLY"
    if x["lower"]>0:return "RESOLVED_POSITIVE"
    if x["upper"]<0:return "RESOLVED_NEGATIVE"
    return "UNRESOLVED_AT_TOLERANCE"

def build_liouvillian(h, jumps):
    eye = sp.eye(h.shape[0], format='csr', dtype=complex)
    result = -1j * (sp.kron(eye, h, format='csr') - sp.kron(h.T, eye, format='csr'))
    for _, c in jumps:
        a = clean(c.getH() @ c)
        result += sp.kron(c.conjugate(), c, format='csr')
        result -= 0.5 * sp.kron(eye, a, format='csr')
        result -= 0.5 * sp.kron(a.T, eye, format='csr')
    return clean(result)

def right(rho, a):

def make_direct_rhs(h, jumps):
    products = [(c, c.getH().tocsr(), clean(c.getH() @ c)) for _, c in jumps]

    def rhs(v):
        rho = v.reshape(h.shape, order='F')
        out = -1j * (h @ rho - right(rho, h))
        for c, ch, a in products:
            out += right(c @ rho, ch) - 0.5 * (a @ rho + right(rho, a))
        return out.reshape(-1, order='F')
    return rhs

def sector_hamiltonian(mask, phi=0., epsilon=0., amplitudes=(1., 1., 1.), theta=None):
    theta = [0., 0., -phi] if theta is None else theta

def null_projector(a):
    _, s, vh = np.linalg.svd(a, full_matrices=True)
    tol = 1e-10*max(1., s[0] if len(s) else 0.)
    v = vh[int(np.sum(s > tol)):].conj().T
    p = v@v.conj().T
    return p, dict(singular_values=s.tolist(), tolerance=tol,
                   dimension=v.shape[1], residual=float(np.linalg.norm(a@p)))

def dark_projectors(h):
    c = np.array([[0., 0., 1.]])
    k, ki = null_projector(np.vstack([c, c@h]))
    d, di = null_projector(np.vstack([c, c@h, c@h@h]))
    connected = {2}
    for _ in range(3):
        connected |= {j for i in list(connected) for j in range(3)
                      if i != j and abs(h[i, j]) > 1e-12}
    isolated = np.diag([float(i not in connected) for i in range(3)])
    # Every sink-disconnected component is invariant; separate it from interference.
    return k, d, isolated, d-isolated, dict(K=ki, D=di)

def compare_pair(a,b):
    # Full common scalar/diagnostic grids, not just the reduced report grid.
    result={};passed=True
    for key in ('series','diagnostics'):
        x,y=a[key],b[key]
        if x.shape!=y.shape or not np.array_equal(x[:,0],y[:,0]):
            return dict(pass_gate=False,reason='grid_mismatch',errors={})
        if not np.array_equal(np.isnan(x),np.isnan(y)):
            return dict(pass_gate=False,reason='conditional_validity_mismatch',errors={})
        finite=np.isfinite(x)&np.isfinite(y)
        error=np.where(finite,abs(x-y),0.)
        scale=1e-6+1e-4*np.maximum(np.nan_to_num(abs(x)),np.nan_to_num(abs(y)))
        # Quality RHS/eigenvalues are still compared, as are all measured diagnostics.
        passed &= bool(np.all(error<=scale))
        names=['time']+(a['names'] if key=='series' else a['dnames'])
        result[key]=dict(zip(names,error.max(axis=0).tolist()))
    return dict(pass_gate=passed,reason=None,errors=result)

def extrema(a):
    data=[]
    for j in range(1,a.shape[1]):
        valid=np.isfinite(a[:,j]);rows=np.flatnonzero(valid)
        if len(rows):
            lo=rows[np.argmin(a[valid,j])];hi=rows[np.argmax(a[valid,j])]
            data.append([float(a[lo,j]),float(a[lo,0]),float(a[hi,j]),float(a[hi,0]),len(rows)])
        else:data.append([None,None,None,None,0])
    return data
