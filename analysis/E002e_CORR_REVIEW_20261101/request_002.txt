"""Fixed-marginal preparation, finite-window propagation and interval decisions.

No projection, clipping, renormalization, inferred infinite tail or Fano label.
The replacement channel has a fixed prepared edge state for each baseline.
"""
import hashlib
import time
import numpy as np
from e002_exp.model import maxabs
from e002e_exp.compat import trapezoid

def marginals(rho, dimensions):
    a, b = dimensions
    rho = np.asarray(rho, complex)
    if rho.shape != (a*b, a*b) or not np.isfinite(rho).all():
        raise ValueError('Invalid bipartite density matrix')
    tensor = rho.reshape(a, b, a, b)
    return np.trace(tensor, axis1=1, axis2=3), np.trace(tensor, axis1=0, axis2=2)


def replacement(rho, edge_reference, dimensions):
    """Linear CPTP only for a FIXED edge_reference.

    prepare chooses a target from its known baseline; the overall map from an
    arbitrary unknown baseline to its two marginals is not a universal channel.
    """
    nodes, _ = marginals(rho, dimensions)
    b = dimensions[1]
    edge_reference = np.asarray(edge_reference, complex)
    if (edge_reference.shape != (b, b) or not np.isfinite(edge_reference).all()
            or maxabs(edge_reference-edge_reference.conj().T) > 1e-10
            or abs(np.trace(edge_reference)-1) > 1e-10
            or np.linalg.eigvalsh(edge_reference).min() < -1e-10):
        raise ValueError('Invalid fixed reference state')
    return np.kron(nodes, edge_reference)


def physicality(space, state, tolerance):
    if not np.isfinite(state).all():
        raise ArithmeticError('Density matrix is nonfinite')
    rho = space.unpack(state)
    herm = maxabs(rho-rho.conj().T)
    minimum = float(np.linalg.eigvalsh((rho+rho.conj().T)/2).min())
    check = dict(trace_error=float(abs(space.trace(state)-1)), hermiticity=herm,
                 minimum_eigenvalue=minimum, psd_mode='FULL_HILBERT')
    if (not np.isfinite(state).all() or check['trace_error'] > tolerance
            or herm > tolerance or minimum < -tolerance):
        raise ArithmeticError('Density matrix physicality failed: '+str(check))
    return check


def array_hash(a):
    a = np.ascontiguousarray(a, dtype='<c16')
    return hashlib.sha256(str(a.shape).encode()+b'\0'+a.tobytes()).hexdigest()


def sector_check(space, rho, product, tolerance):
    """Certify this builder's complete q=0 block algebra, with no projection.

    The 216-state model has Q=N_nodes tensor I_edges. For [rho,Q]=0,
    [Tr_edges(rho),N_nodes]=0, hence [rho_nodes tensor rho_edges,Q]=0.
    No per-edge superselection rule from an older 512-state model is assumed.
    """
    joined = np.concatenate(space.groups)
    if not np.array_equal(np.sort(joined), np.arange(space.d)):
        raise ValueError('Charge groups do not partition full Hilbert space')
    outside = space.charges[:, None] != space.charges[None, :]
    errors = dict(input_off_sector=maxabs(np.where(outside, rho, 0)),
                  product_off_sector=maxabs(np.where(outside, product, 0)),
                  input_roundtrip=maxabs(space.unpack(space.pack(rho))-rho),
                  product_roundtrip=maxabs(space.unpack(space.pack(product))-product))
    if max(errors.values()) > tolerance:
        raise ArithmeticError('Preparation violates q=0 sector/roundtrip')
    return errors


def prepare(model, state, dimensions=(8, 27), tolerance=1e-10):
    space = model['space']; rho = space.unpack(state)
    before = physicality(space, state, tolerance)
    nodes, edges = marginals(rho, dimensions)
    product = replacement(rho, edges, dimensions)
    sector = sector_check(space, rho, product, tolerance)
    cut = space.pack(product)
    after = physicality(space, cut, tolerance)
    # Verify the actual packed/unpacked state, not just the formal kron target.
    new_nodes, new_edges = marginals(space.unpack(cut), dimensions)
    errors = dict(nodes=maxabs(nodes-new_nodes), edges=maxabs(edges-new_edges))
    if max(errors.values()) > tolerance:
        raise ArithmeticError('Preparation changed a marginal')
    change = space.unpack(cut)-rho
    distance = .5*float(np.abs(np.linalg.eigvalsh((change+change.conj().T)/2)).sum())
    record = dict(marginal_errors=errors, sector_errors=sector,
                  sector_contract='COMPLETE_Q0_CHARGE_BLOCKS_NO_PROJECTION',
                  preparation_type='PRODUCT_OF_KNOWN_BASELINE_MARGINALS_FIXED_TARGET',
                  universal_cptp_channel=False,
                  hashes={n:array_hash(a) for n,a in (('state',state),('cut',cut),('nodes',nodes),('edges',edges))},
                  marginal_spectra=dict(nodes=np.linalg.eigvalsh((nodes+nodes.conj().T)/2).tolist(),
                                        edges=np.linalg.eigvalsh((edges+edges.conj().T)/2).tolist()),
                  sham_physicality=before, cut_physicality=after,
                  trace_distance=distance, frobenius_distance=float(np.linalg.norm(change)),
                  delta_energy=float(np.trace(model['H']@change).real),
                  external_preparation_cost='NOT_ASSUMED_ZERO_NOT_MEASURED',
                  removed_correlations='ALL_NODE_EDGE_CORRELATIONS_NOT_ONLY_ENTANGLEMENT')
    return np.column_stack((state, cut)), record


def grid_steps(horizon, dt, interval):
    if not np.isfinite([horizon, dt, interval]).all() or min(horizon, dt, interval) <= 0:
        raise ValueError('Invalid finite time grid')
    n, stride = round(horizon/dt), round(interval/dt)
    if n < 1 or stride < 1 or abs(n*dt-horizon) > 1e-10 or abs(stride*dt-interval) > 1e-10 or n % stride:
        raise ValueError('Grid must contain exact integral steps and endpoint')
    return n, stride


def propagate(model, initial, *, horizon, dt, sample_interval, physicality_interval,
              tolerance=1e-9, checkpoint=None, density_checks=True):
    """Two density columns (sham/cut), RK4; retain flux at EVERY step.

    Other observables are retained on the declared common grid. Physicality
    is checked at the declared checkpoints, not claimed between checkpoints.
    density_checks=False is reserved for synthetic calibration vectors.
    """
    steps, stride = grid_steps(horizon, dt, sample_interval)
    _, check_stride = grid_steps(horizon, dt, physicality_interval)
    space = model['space']; L = model['L']; y = np.asarray(initial, complex).copy()
    if y.shape != (space.size, 2) or not np.isfinite(y).all():
        raise ValueError('Expected two finite state columns')
    names = tuple(model['observables']); rows = np.vstack([space.row(model['observables'][n]) for n in names])
    frow = space.row(model['observables']['F'])
    times = np.arange(steps+1)*dt; flux = np.empty((steps+1, 2))
    output = []; checks = []; trace0 = space.trace(y)
    start = time.perf_counter()
    for step in range(steps+1):
        if step:
            a = L@y; b = L@(y+dt*a/2); c = L@(y+dt*b/2); d = L@(y+dt*c)
            y += dt*(a+2*b+2*c+d)/6
        f = frow@y
        if not np.isfinite(y).all() or maxabs(f.imag) > tolerance:
            raise ArithmeticError('Evolution is nonfinite or flux is complex')
        flux[step] = f.real
        if step % stride == 0:
            output.append(rows@y)
        if step % check_stride == 0:
            if maxabs(space.trace(y)-trace0) > tolerance:
                raise ArithmeticError('Evolution trace drift')
            if density_checks:
                checks.append(dict(time=float(times[step]), states=[physicality(space, y[:, k], tolerance) for k in range(2)]))
            if checkpoint:
                checkpoint(step, float(times[step]), y)
    return dict(time=times, flux=flux, sample_time=times[::stride], raw=np.asarray(output), names=names,
                time_index=np.arange(steps+1), sample_index=np.arange(steps+1)[::stride], dt=float(dt),
                final=y, checks=checks, steps=steps, checkpoint_calls=steps//check_stride+1,
                block_rhs_calls=4*steps,
                scalar_rhs_equivalents=8*steps, elapsed_seconds=time.perf_counter()-start)


def threshold_decision(value, uncertainty, threshold):
    if (not np.isfinite([value, uncertainty, threshold]).all()
            or value < 0 or uncertainty < 0 or threshold <= 0):
        raise ValueError('Invalid nonnegative interval')
    lo, hi = max(0., value-uncertainty), value+uncertainty
    status = ('MARGINAL_SUFFICIENCY_REJECTED_AT_DECLARED_RESOLUTION' if lo > threshold else
              'MARGINAL_SUFFICIENCY_NOT_REJECTED_IN_WINDOW' if hi < threshold else 'UNRESOLVED')
    return dict(value=value, uncertainty=uncertainty, interval=[lo, hi], threshold=threshold, status=status)


def analyze(fine, coarse, reference, *, baseline_flux, reference_flux, policy):
    for run in (fine, coarse, reference):
        if not np.isfinite(run['flux']).all():
            raise ValueError('Nonfinite trajectory')
    def same_grid(a,b):
        return a.shape == b.shape and np.allclose(a,b,rtol=0.,atol=1e-12)
    if (not same_grid(fine['time'], reference['time'])
            or not same_grid(fine['sample_time'], coarse['sample_time'])
            or not same_grid(fine['sample_time'], reference['sample_time'])
            or fine['names'] != coarse['names'] or fine['names'] != reference['names']):
        raise ValueError('Comparison grids or observables differ')
    if abs(fine['time'][-1]-coarse['time'][-1]) > 1e-12:
        raise ValueError('Comparison horizons differ')
    if baseline_flux <= 1e-6 or reference_flux <= 1e-6:
        raise ValueError('Unresolved normalization flux')
    # Fixed numerical caps are independent of the scientific threshold.
    comparisons = {}
    for name, other in (('dt', coarse), ('initial_condition_and_solver_sensitivity', reference)):
        err = abs(fine['raw']-other['raw'])
        scale = policy['raw_absolute_tolerance']+policy['raw_relative_tolerance']*np.maximum(abs(fine['raw']), abs(other['raw']))
        comparisons[name] = dict(max_abs=float(err.max()), max_scaled=float((err/scale).max()), passed=bool(np.all(err <= scale)))
    if not all(x['passed'] for x in comparisons.values()):
        raise ArithmeticError('Trajectory refinement failed: '+str(comparisons))
    t = fine['time']; H = float(t[-1]); delta = fine['flux'][:, 1]-fine['flux'][:, 0]
    dc = coarse['flux'][:, 1]-coarse['flux'][:, 0]; dr = reference['flux'][:, 1]-reference['flux'][:, 0]
    u = policy['absolute_floor']+policy['safety']*(abs(delta-np.interp(t, coarse['time'], dc))+abs(delta-dr))
    area = float(trapezoid(abs(delta), t)); u_area = float(trapezoid(u, t))
    uf = policy['absolute_floor']+policy['safety']*abs(baseline_flux-reference_flux)
    R = ratio(area, H*baseline_flux, u_area, H*uf)
    if R['value'] is None or R['uncertainty'] is None or R['uncertainty'] > policy['maximum_metric_uncertainty']:
        raise ArithmeticError('Primary uncertainty is unresolved or exceeds fixed cap')
    verdict = threshold_decision(R['value'], R['uncertainty'], policy['primary_threshold'])
    raw_error = policy['absolute_floor']+policy['safety']*(abs(fine['raw']-coarse['raw'])+abs(fine['raw']-reference['raw']))
    h = fine['names'].index('h6'); p = fine['names'].index('P_m6_N2')
    conditional = []
    for i, ti in enumerate(fine['sample_time']):
        row = []
        for k in range(2):
            hh, pp = fine['raw'][i, h, k], fine['raw'][i, p, k]
            if max(abs(hh.imag), abs(pp.imag)) > policy['raw_absolute_tolerance']:
                raise ArithmeticError('Conditional primitive is complex')
            row.append(ratio(float(hh.real), float(pp.real), float(raw_error[i, h, k]), float(raw_error[i, p, k])))
        conditional.append(dict(time=float(ti), sham=row[0], cut=row[1]))
    peak = int(np.argmax(abs(delta)))
    sham_drift = float(np.max(abs(fine['flux'][:, 0]-baseline_flux)))
    reference_sham_drift = float(np.max(abs(reference['flux'][:,0]-reference_flux)))
    coarse_sham_drift = float(np.max(abs(coarse['flux'][:,0]-baseline_flux)))
    if max(sham_drift, reference_sham_drift, coarse_sham_drift) > policy['maximum_sham_drift']:
        raise ArithmeticError('Sham stationary drift exceeds fixed cap')
    return dict(primary=verdict, area_absolute=area, area_uncertainty=u_area,
                signed_area=float(trapezoid(delta, t)), peak_relative=float(abs(delta[peak])/baseline_flux),
                time_peak=float(t[peak]), final_delta_flux=float(delta[-1]), sham_drift=sham_drift,
                reference_sham_drift=reference_sham_drift, coarse_sham_drift=coarse_sham_drift,
                refinement=comparisons,
                uncertainty_kind='ENGINEERING_DT_AND_INITIAL_CONDITION_SOLVER_SENSITIVITY_NOT_RIGOROUS_TOTAL_BOUND',
                tail_status='FINITE_WINDOW_ONLY', conditional=conditional, raw_uncertainty=raw_error)

        blob = hashlib.sha1(b'blob '+str(len(raw)).encode()+b'\0'+raw).hexdigest()

    baselines = []; canonical = []

                            repository_commit=pins['repository_ref'])

                        {k:obj[k] for k in ('families','lines','width_status','background_status')} if tag == 'F2' else
                        {k:obj[k] for k in ('fits','width_transfer','exact_ratio_decomposition')})
    evidence = dict(schema='E002e_CORR_evidence_v1', protocol_sha256=protocol_hash(), baselines=baselines, sources=sources,

    dump = lambda p,v: p.write_text(json.dumps(v, ensure_ascii=False, sort_keys=True, indent=2, allow_nan=False)+'\n')

    if not pools or any(p.get('num_threads') != 1 for p in pools):
        raise ValueError('Measured BLAS/OpenMP thread count must be one')
    return dict(status='PASS', requested=requested, measured=pools)


def storage_preflight(destination):
    # Retained data: 3 initial pairs + 9 rolling checkpoints (<8 MB raw),
    # full sampled series ~12 MB, JSON diagnostics/manifests. Bound also
    # covers one atomic temporary file; no 30000-checkpoint history is kept.
    required = 128*2**20; reserve = 2**30
    available = shutil.disk_usage(destination).free
    if available < required+reserve:
        raise OSError('Insufficient report/checkpoint storage reserve')
    return dict(status='PASS', required_bytes=required, reserve_bytes=reserve,
                available_bytes=available, retention='LAST_ONLY_NO_RESUME')


def cases():
    out = []
    for name in MODELS:
        matches = [c for c in selected_work('baseline')['full'] if c['model'] == name
                   and c['phi_over_pi'] == .5 and c['epsilon'] == 0 and c['node_noise'] == .05
                   and c['m6_v_over_J'] == PROTOCOL['m6_v_over_J']]
        if len(matches) != 1:
            raise ValueError('Baseline identity ambiguous')
        out.append(dict(matches[0], tasks=['NESS']))
    return out


def workload(horizon=20.):
    fine = round(horizon/PROTOCOL['dt_fine']); coarse = round(horizon/PROTOCOL['dt_coarse'])
    return dict(NESS=3, refinement_NESS=3, propagations=9, block_steps=3*(2*fine+coarse),
                checkpoint_writes=9*(round(horizon/PROTOCOL['checkpoint_interval'])+1),
                block_rhs_calls=12*(2*fine+coarse), scalar_rhs_equivalents=24*(2*fine+coarse))


def preflight(config, current):
    mode = config.get('mode')
    if mode not in ('calibration', 'production'):
        raise ValueError('Explicit supported mode required')
    if config.get('protocol_sha256') != protocol_hash() or config.get('implementation_sha256') != implementation_hash():
        raise ValueError('Unpinned or changed protocol/implementation')
    source = config.get('source_commit')
    if not re.fullmatch('[0-9a-f]{40}', source or '') or os.environ.get('VM_SOURCE_COMMIT') != source:
        raise ValueError('VM checkout/source identity mismatch')
    # These settings are fixed in code; configuration cannot relax their caps.
    allowed = {'mode', 'protocol_sha256', 'implementation_sha256', 'source_commit',
               'technical_review', 'evidence', 'calibration', 'forecast', 'memory_budget'}
    if set(config)-allowed:
        raise ValueError('Unsupported configuration override')
    threads = thread_preflight(current)
    review = pinned_file(config['technical_review'])
    if (review.get('status') != 'REVIEWED_AND_CHECKED' or review.get('implementation_sha256') != implementation_hash()
            or not re.fullmatch('[0-9a-f]{40}', review.get('reviewed_source_commit', ''))):
        raise ValueError('Technical review missing for current implementation')
    evidence = pinned_file(config['evidence'])
    if evidence.get('schema') != 'E002e_CORR_evidence_v1' or evidence.get('protocol_sha256') != protocol_hash():
        raise ValueError('Evidence/protocol mismatch')
    records = {}
    for item in evidence['baselines']:
        record = pinned_file(item)
        if item['case_id'] in records or record['id'] != item['case_id']:
            raise ValueError('Duplicate/mismatched evidence')
        records[item['case_id']] = record
    if set(records) != {c['id'] for c in cases()}:
        raise ValueError('Incomplete baseline evidence')
    for case in cases():
        record = records[case['id']]
        if any(record['case'][k] != case[k] for k in ('model', 'epsilon', 'phi_over_pi', 'node_noise', 'm6_v_over_J')):
            raise ValueError('Baseline physics mismatch')
        if not record['quality']['pass'] or not record['refinement']['gate']['pass_gate']:
            raise ValueError('Unverified published baseline')
    horizon = PROTOCOL['horizon'] if mode == 'production' else PROTOCOL['calibration_horizon']
    if mode == 'production':
        cal = pinned_file(config['calibration'])
        if (cal.get('status') != 'CALIBRATION_COMPLETE' or cal.get('runtime_review_status') != 'REVIEWED'
                or not re.fullmatch('[0-9a-f]{40}', cal.get('report_commit', ''))
                or cal.get('implementation_sha256') != implementation_hash()
                or cal.get('protocol_sha256') != protocol_hash() or cal.get('counts') != workload(PROTOCOL['calibration_horizon'])):
            raise ValueError('Calibration identity/coverage mismatch')
        env = compare_environment(cal['environment'], current)
        if not env['comparable']:
            raise ValueError('Production environment differs from measured calibration')
        if config['forecast'] != cal['production_forecast'] or config['memory_budget'] != cal['memory_budget']:
            raise ValueError('Budget is not the pinned measured budget')
    else:
        env = dict(comparable=False, reason='BOUNDED_FIRST_CALIBRATION_NO_CORRELATION_CUT')
    f = config['forecast']
    if (not all(np.isfinite(f[k]) for k in ('expected_seconds', 'lower_seconds', 'upper_seconds'))
            or not 0 < f['lower_seconds'] <= f['expected_seconds'] <= f['upper_seconds']):
        raise ValueError('Invalid runtime forecast')
    memory = memory_preflight(config['memory_budget'], current)
    return dict(cases=cases(), evidence=records, environment=env, memory=memory, threads=threads,
                counts=workload(horizon), horizon=horizon)

def save(path, value):
    Path(path).write_text(json.dumps(serial(value), ensure_ascii=False, indent=2, allow_nan=False)+'\n')


def select_observables(model):
    names = ('F', 'J_h', 'source', 'inventory', 'h6', 'P_m6_N2', 'K')
    result = {n:model['observables'][n] for n in names}
    result.update({n:o for n,o in model['observables'].items() if n.startswith('C_')})
    for i, op in enumerate(model['n']):
        result['node_'+str(i)] = op
    for i, op in enumerate(model['currents']):
        result['current_'+str(i)] = op
    # Preserve the public model builder; local marginal diagnostics only.
    if model['space'].d == 216:
        o = operators()
        for i, op in enumerate(o['pl']):
            if op.shape != (216,216):
                raise ValueError('edge_L operator must already be lifted by the pinned builder')
            result['edge_L_'+str(i)] = op
        for m in range(8):
            result['mask_'+str(m)] = sp.kron(sp.eye(8), o['masks'][m], format='csr')
    if any(op.shape != (model['space'].d,model['space'].d) for op in result.values()):
        raise ValueError('Observable dimension mismatch')
    return dict(model, observables=result)


def output_balance(model):
    if model['space'].d != 216:
        return dict(status='SYNTHETIC_MODEL_NOT_FULL_OUTPUT_BALANCE')
    sinks = [c for name,c in model['jumps'] if name == 'sink_2']
    if len(sinks) != 1:
        raise ValueError('Missing/ambiguous sink operator')
    n = model['n'][2]; loss = sinks[0].getH()@sinks[0]
    kappa = float(loss.diagonal().sum().real/n.diagonal().sum().real)
    F = model['observables']['F']; J = model['observables']['J_h']; H = model['H']
    adj = 1j*(H@F-F@H)+sum(adjoint_jump(c,F) for _,c in model['jumps'])
    residual = max(maxabs(loss-kappa*n),maxabs(F-kappa*n),maxabs(adj-kappa*(J-F)))
    if kappa <= 0 or residual > PROTOCOL['operator_tolerance']:
        raise ArithmeticError('Full output-balance operator identity failed')
    return dict(status='OPERATOR_IDENTITY_VERIFIED',kappa_out=kappa,residual=residual)


def preparation_diagnostics(model, initial, baseline, prep, balance):
    initial_values = real_values(values(model,initial[:,1]))
    if abs(initial_values['F']-baseline['F']) > PROTOCOL['operator_tolerance']:
        raise ArithmeticError('Marginal-preserving intervention changed initial flux')
    derivative = model['L']@(initial[:,1]-initial[:,0])
    df = complex(model['space'].row(model['observables']['F'])@derivative)
    if abs(df.imag) > PROTOCOL['operator_tolerance']:
        raise ArithmeticError('Initial flux derivative is complex')
    prep.update(delta_flux_derivative_at_zero=float(df.real),
                delta_h_at_zero=initial_values['J_h']-baseline['J_h'],
                initial_observables=initial_values, output_balance=balance)
    if balance['status'] == 'OPERATOR_IDENTITY_VERIFIED':
        residual = abs(df.real-balance['kappa_out']*prep['delta_h_at_zero'])
        if residual > PROTOCOL['operator_tolerance']:
            raise ArithmeticError('Initial output balance identity failed')
        prep['initial_balance_residual'] = float(residual)


def atomic_npz(path, **arrays):
    path = Path(path); temp = path.with_name(path.stem+'.tmp.npz')
    try:
        np.savez_compressed(temp, **arrays)
        os.replace(temp,path)
    finally:
        temp.unlink(missing_ok=True)


def solve_baseline(model):
    solutions = []; checks = []; metrics = []; prim = []
    started = time.perf_counter()
    for options in SOLVERS:
        solver = Bordered(model['L'], model['space'], **options)
        state, equation = solver.solve(trace=1.)
        quality = state_checks(model['L'], model['space'], state, PROTOCOL['state_tolerance'])
        if not quality['pass']:
            raise ArithmeticError('NESS quality failed')
        solutions.append(state); prim.append(real_values(values(model, state)))
        checks.append(dict(equation=equation, quality=quality)); metrics.append(solver.metrics())
        del solver
    comparison = refinement_check(*solutions, *prim, LIMITS)
    if not comparison['pass_gate']:
        raise ArithmeticError('Independent NESS methods disagree')
    for raw in prim:
        if abs(raw['F']-raw['J_h']) > 1e-9 or abs(raw['F']-raw['source']) > 1e-9:
            raise ArithmeticError('Stationary flux balance failed')
    return solutions, prim, dict(checks=checks, comparison=comparison, solvers=metrics,
                                  seconds=time.perf_counter()-started)


def run_case(model, states, raw, *, mode, horizon, output=None, dimensions=(8, 27)):
    """Scientific routine also tested on independently solvable small models."""
    preparation_started = time.perf_counter()
    balance = output_balance(model)
    if mode == 'production':
        initial, prep = prepare(model, states[0], dimensions, PROTOCOL['operator_tolerance'])
        initial_ref, prep_ref = prepare(model, states[1], dimensions, PROTOCOL['operator_tolerance'])
        for x,b,p in ((initial,raw[0],prep),(initial_ref,raw[1],prep_ref)):
            preparation_diagnostics(model,x,b,p,balance)
    elif mode == 'calibration':
        initial = np.column_stack((states[0], states[0]))
        initial_ref = np.column_stack((states[1], states[1]))
        prep = dict(status='KNOWN_BASELINE_SHAM_ONLY_NO_NEW_INTERVENTION')
        prep_ref = prep.copy()
    else:
        raise ValueError('Unsupported mode')
    if output:
        atomic_npz(output/'initial.npz', primary=initial, independent=initial_ref)
    preparation_seconds = time.perf_counter()-preparation_started
    runs = []
    for label, start, dt in (('fine', initial, PROTOCOL['dt_fine']),
                             ('coarse', initial, PROTOCOL['dt_coarse']),
                             ('reference', initial_ref, PROTOCOL['dt_fine'])):
        io = dict(seconds=0.,writes=0,bytes_written=0)
        def checkpoint(step, t, y):
            if output:
                tick = time.perf_counter(); path = output/(label+'_checkpoint.npz')
                atomic_npz(path,state=y,step=step,time=t)
                io['seconds'] += time.perf_counter()-tick
                io['writes'] += 1; io['bytes_written'] += path.stat().st_size
        runs.append(propagate(model, start, horizon=horizon, dt=dt,
                              sample_interval=PROTOCOL['sample_interval'],
                              physicality_interval=PROTOCOL['physicality_interval'],
                              tolerance=PROTOCOL['state_tolerance'], checkpoint=checkpoint))
        runs[-1]['checkpoint_io'] = io
    result = analyze(*runs, baseline_flux=raw[0]['F'], reference_flux=raw[1]['F'], policy=PROTOCOL)
    if mode == 'calibration':
        result.pop('primary')
        result['scientific_status'] = 'NO_INTERVENTION_VERDICT_DURING_CALIBRATION'
    result.update(preparation=prep, reference_preparation=prep_ref, preparation_seconds=preparation_seconds)
    return runs, result

def run(config, output):
    output = Path(output)
    if (output/'report').exists() or (output/'full').exists():
        raise FileExistsError('Refusing to overwrite existing run')
    output.mkdir(parents=True, exist_ok=True); started = time.perf_counter()
    try:
        return _run(config, output)
    except Exception as exc:
        report = output/'report'; report.mkdir(exist_ok=True)
        save(report/'failure.json', dict(type=type(exc).__name__, message=str(exc),
             elapsed_seconds=time.perf_counter()-started, traceback=traceback.format_exc()))
        save(report/'status.json', dict(status='FAILED', scientific_acceptance=False))

def numeric_equal(a,b,atol=1e-13,rtol=1e-12):
    if isinstance(a,dict) and isinstance(b,dict):
        return a.keys()==b.keys() and all(numeric_equal(a[k],b[k],atol,rtol) for k in a)
    if isinstance(a,list) and isinstance(b,list):
        return len(a)==len(b) and all(numeric_equal(x,y,atol,rtol) for x,y in zip(a,b))
    if isinstance(a,bool) or isinstance(b,bool) or a is None or b is None:
        return type(a) is type(b) and a==b
    if isinstance(a,(int,float)) and isinstance(b,(int,float)):
        return bool(np.isfinite([a,b]).all() and np.isclose(a,b,atol=atol,rtol=rtol))
    return a==b


def verify_full_files(report,full):
    """Optional local VM audit; never fetches an unpublished archive."""
    full = Path(full)
    manifest = json.loads((report/'full_manifest.json').read_text())
    expected = {e['path'] for e in manifest['files']}
    if len(expected) != len(manifest['files']) or expected != {p.relative_to(full).as_posix() for p in full.rglob('*') if p.is_file()}:
        raise ValueError('Full archive coverage mismatch')
    for e in manifest['files']:
        rel = Path(e['path'])
        if rel.is_absolute() or '..' in rel.parts or '.tmp' in rel.name:
            raise ValueError('Unsafe full archive path')
        data = (full/rel).read_bytes()
        if len(data)!=e['size'] or hashlib.sha256(data).hexdigest()!=e['sha256']:
            raise ValueError('Full archive checksum mismatch')
    if manifest['total_bytes'] != sum(e['size'] for e in manifest['files']):
        raise ValueError('Full archive byte total mismatch')

def classical_model():
    # Correlated classical four-state process is an essential positive control:
    # CORR must NOT mistake a signal here for a quantum-specific mechanism.
    space = Space(np.zeros(4, int)); H = sp.csr_matrix((4,4), dtype=complex)
    jumps = []
    for a,b,rate in ((0,2,1.),(2,0,.1),(1,3,.2),(3,1,1.4),
                     (0,1,.3),(1,0,.3),(2,3,.3),(3,2,.3)):
        C = sp.csr_matrix(([np.sqrt(rate)], ([b],[a])), shape=(4,4), dtype=complex)
        jumps.append((f'{a}_{b}',C))
    na = sp.diags([0.,0.,1.,1.]); nb = sp.diags([0.,1.,0.,1.]); F = .5*na
    obs = dict(F=F,J_h=F,source=F,inventory=nb,h6=F,P_m6_N2=sp.eye(4),K=na-nb)
    return dict(space=space,H=H,jumps=jumps,L=liouvillian(H,jumps,space),observables=obs,
                n=[na,nb],currents=[F,0*F,0*F])


def trajectories(model, states, H=1.):

    return [propagate(model,x,horizon=H,dt=dt,sample_interval=.02,physicality_interval=.5)
            for x,dt in ((initial,.005),(initial,.01),(reference,.005))]


class PreparationTests(unittest.TestCase):
    def setUp(self):
        v = np.array([0.,1.,1.,0.])/np.sqrt(2)
        self.rho=.8*np.outer(v,v)+.2*np.eye(4)/4

    def test_full_complex_marginals_not_only_diagonals(self):
        U = np.array([[1,1j],[1j,1]])/np.sqrt(2)
        rho = np.kron(U,np.eye(2))@self.rho@np.kron(U.conj().T,np.eye(2))
        a,b=marginals(rho,(2,2)); prod=replacement(rho,b,(2,2))
        aa,bb=marginals(prod,(2,2))
        np.testing.assert_allclose(a,aa,atol=1e-14);np.testing.assert_allclose(b,bb,atol=1e-14)
        self.assertGreater(np.linalg.norm(prod-rho),.1)
        self.assertGreaterEqual(np.linalg.eigvalsh(prod).min(),0.)

    def test_fixed_reference_channel_is_linear(self):
        tau=np.diag([.3,.7]); other=np.diag([.1,.2,.3,.4])
        np.testing.assert_allclose(replacement(.4*self.rho+.6*other,tau,(2,2)),
                                   .4*replacement(self.rho,tau,(2,2))+.6*replacement(other,tau,(2,2)),atol=1e-14)

    def test_product_is_fixed_point(self):
        a=np.array([[.6,.1j],[-.1j,.4]]);b=np.array([[.4,.2],[.2,.6]])
        np.testing.assert_allclose(replacement(np.kron(a,b),b,(2,2)),np.kron(a,b),atol=1e-14)

    def test_invalid_reference_rejected(self):
        for tau in (np.diag([1.1,-.1]),np.eye(2),np.full((2,2),np.nan)):

"""Charge-covariant q=0 Liouvillian, without a dense Liouville matrix."""
from dataclasses import dataclass
import numpy as np
from scipy import sparse as sp
from e002_exp.model import clean, maxabs


@dataclass
class Space:
    charges: np.ndarray

    def __post_init__(self):
        self.charges = np.asarray(self.charges, dtype=int)
        self.d = len(self.charges)
        self.groups = [np.flatnonzero(self.charges == n) for n in np.unique(self.charges)]
        self.sizes = [len(g) ** 2 for g in self.groups]
        self.offsets = np.r_[0, np.cumsum(self.sizes)]
        self.indices = np.concatenate([(g[:, None] + self.d*g[None, :]).ravel(order='F') for g in self.groups])
        self.trace_indices = np.concatenate([off + np.arange(len(g))*(len(g)+1)
                                            for off, g in zip(self.offsets, self.groups)])
        self.size = len(self.indices)
        self.trace_row = sp.csr_matrix((np.ones(self.d), (np.zeros(self.d, int), self.trace_indices)), shape=(1, self.size))

    def pack(self, matrix, tolerance=1e-12):
        a = matrix.toarray() if sp.issparse(matrix) else np.asarray(matrix)
        if a.shape != (self.d, self.d):
            raise ValueError('Hilbert dimensions disagree')
        if maxabs(np.where(self.charges[:, None] != self.charges[None, :], a, 0)) > tolerance:
            raise ValueError('Input has components outside q=0')
        return a.ravel(order='F')[self.indices].copy()

    def unpack(self, vector):
        v = np.asarray(vector)
        if v.shape != (self.size,):
            raise ValueError('Reduced vector shape mismatch')
        a = np.zeros(self.d*self.d, complex)
        a[self.indices] = v
        return a.reshape(self.d, self.d, order='F')

    def row(self, operator):
        # trace(O rho), without conjugation of O: vec(O.T)^T vec(rho).
        a = operator.toarray() if sp.issparse(operator) else np.asarray(operator)
        return a.T.ravel(order='F')[self.indices]

    def trace(self, vector):
        return np.asarray(vector)[self.trace_indices].sum(axis=0)


def charge_checks(H, jumps, space, tolerance=1e-12):
    h = sp.coo_matrix(H)
    hbad = maxabs(h.data[space.charges[h.row] != space.charges[h.col]])
    deltas = {}
    for name, C in jumps:
        c = sp.coo_matrix(C)
        nz = abs(c.data) > tolerance
        ds = np.unique(space.charges[c.row[nz]] - space.charges[c.col[nz]])
        if len(ds) > 1:
            raise ValueError('Jump lacks definite charge: ' + name)
        deltas[name] = int(ds[0]) if len(ds) else None
    if hbad > tolerance:
        raise ValueError('Hamiltonian violates charge covariance')
    return dict(H_charge_residual=hbad, jump_charge_changes=deltas)


def liouvillian(H, jumps, space):
    """Exact block construction; every within-charge coherence is retained."""
    charge_checks(H, jumps, space)
    H = sp.csr_matrix(H)
    count = len(space.groups)
    blocks = [[sp.csr_matrix((space.sizes[a], space.sizes[b]), dtype=complex)
               for b in range(count)] for a in range(count)]
    for a, g in enumerate(space.groups):
        h = H[g][:, g]; eye = sp.eye(len(g), format='csr')
        blocks[a][a] += -1j*(sp.kron(eye, h) - sp.kron(h.T, eye))
    for _, C in jumps:
        C = sp.csr_matrix(C); loss = clean(C.getH() @ C)
        for a, ga in enumerate(space.groups):
            eye = sp.eye(len(ga), format='csr'); m = loss[ga][:, ga]
            blocks[a][a] -= .5*(sp.kron(eye, m) + sp.kron(m.T, eye))
            for b, gb in enumerate(space.groups):
                c = C[ga][:, gb]
                if c.nnz:
                    blocks[a][b] += sp.kron(c.conjugate(), c)
    L = clean(sp.bmat(blocks, format='csr'))
    if maxabs(space.trace_row @ L) > 1e-11:
        raise ValueError('Reduced generator is not trace preserving')
    return L


def dissipator_action(C, rho):
    C = sp.csr_matrix(C); m = C.getH() @ C
    right = lambda a, b: (b.T @ a.T).T
    return right(C @ rho, C.getH()) - .5*(m @ rho + right(rho, m))

"""Physical operators for the approved geometry ladder and full model."""
import numpy as np
from scipy import sparse as sp
from e002_exp.model import build_model, clean, ketbra, tensor, maxabs, adjoint_jump
from e002c_exp.model import BASE, RR, EDGES, PAIRS, operators
from e002d_exp.plan import TARGETS, OPERATIONS
from .space import Space, liouvillian, charge_checks

IC_ALPHA = (.06489310345717117, .06489313511138023, .07065096360073866)


def node_ops():
    eye = sp.eye(8, dtype=complex, format='csr')
    def emb(x, at):
        return tensor([x if k == at else sp.eye(2, format='csr') for k in range(3)])
    up = [emb(ketbra(2, 1, 0), k) for k in range(3)]
    n = [x @ x.getH() for x in up]
    return eye, up, n


def make_small(case):
    d = int(case['dimension']); mask = int(case['mask'])
    if d not in (4, 8) or mask not in (6, 7):
        raise ValueError('Unsupported ladder case')
    source = float(case.get('source', .5)); sink = float(case.get('sink', .5))
    noise = float(case.get('node_noise', .05)); v = float(case.get('v_over_J', 1.))
    eps = float(case['epsilon']); phi = float(case['phi_over_pi'])*np.pi
    if min(source, sink) <= 0 or min(noise, v) < 0 or not np.isfinite([source, sink, noise, v, eps, phi]).all():
        raise ValueError('Invalid small-model parameters')
    eye, up8, n8 = node_ops()
    if d == 4:
        # Order g, node0, node1, node2; independent global N<=1 model.
        eye = sp.eye(4, format='csr', dtype=complex)
        up = [ketbra(4, i+1, 0) for i in range(3)]
        n = [ketbra(4, i+1, i+1) for i in range(3)]
        charges = np.array([0, 1, 1, 1])
        hops = [ketbra(4, i+1, j+1) for i, j in PAIRS]
    else:
        up, n = up8, n8
        charges = np.array([k.bit_count() for k in range(8)])
        hops = [up[i] @ up[j].getH() for i, j in PAIRS]
    H = eps*n[1]; currents = []
    for e, ((i, j), a, theta) in enumerate(zip(PAIRS, hops, (0., 0., -phi))):
        amp = v if e == (1 if mask == 6 else 0) else 1.
        h = amp*(np.exp(1j*theta)*a + np.exp(-1j*theta)*a.getH()) if mask & (1 << e) else 0*a
        H += h; currents.append(clean(1j*(h@n[j]-n[j]@h)))
    jumps = [('source_0', np.sqrt(source)*up[0]), ('sink_2', np.sqrt(sink)*up[2].getH())]
    convention = case.get('jump_convention', 'three_sqrt_phi_Z')
    if convention == 'sqrt(phi)*n1_only':
        jumps.append(('node_dephase_1', np.sqrt(noise)*n[1]))
    elif convention == 'three_sqrt_phi_Z':
        jumps += [(f'node_dephase_{i}', np.sqrt(noise)*(eye-2*x)) for i, x in enumerate(n)]
    else:
        raise ValueError('Unknown small noise convention')
    active = np.arange(d)
    branch = 'empty_initial_state_unique_stationary_candidate'
    if mask == 6 and v == 0:
        active = np.flatnonzero(abs(n[1].diagonal()) < .5)
        branch = 'n1_zero_invariant_from_empty_initial_state'
        outside = np.setdiff1d(np.arange(d), active)
        if maxabs(H[outside][:, active]) > 1e-12 or any(maxabs(c[outside][:, active]) > 1e-12 for _, c in jumps):
            raise ValueError('Declared physical branch is not invariant')
    restrict = lambda a: clean(a[active][:, active])
    H = restrict(H); jumps = [(name, restrict(c)) for name, c in jumps]
    n = [restrict(x) for x in n]; currents = [restrict(x) for x in currents]
    space = Space(charges[active]); charges = charges[active]
    obs = {'J_h': currents[1]+currents[2], 'F': sink*n[2], 'p_e': sp.diags((charges == 1).astype(float)),
           'P_N2': sp.diags((charges == 2).astype(float)), 'K': currents[1]-currents[2],
           'source': source*(sp.eye(len(active))-sum(n)) if d == 4 else source*(sp.eye(len(active))-n[0])}
    for N in range(4):
        p = sp.diags((charges == N).astype(float), format='csr')
        obs[f'p_N{N}'] = p; obs[f'h_N{N}'] = clean(p@obs['J_h'])
    return dict(H=H, jumps=jumps, space=space, observables=obs, currents=currents,
                n=n, case=case, original_dimension=d, active=active, branch=branch,
                L=liouvillian(H, jumps, space))


def make_full(case, assemble=True):
    from .plan import full_physical_key
    full_physical_key(case)
    o = operators(); model = case['model']; p = dict(BASE, phi_n=float(case['node_noise']))
    if model == 'RR_QS':
        p.update(g=0., gamma_r=0., rr_rates=RR)
    elif model == 'IC1_QS':
        p['g'] = 0.
    elif model != 'QS':
        raise ValueError('Unknown full model')
    phi = float(case['phi_over_pi'])*np.pi; eps = float(case['epsilon']); v = float(case['m6_v_over_J'])
    if not np.isfinite([phi, eps, v, p['phi_n']]).all() or min(v, p['phi_n']) < 0:
        raise ValueError('Invalid full parameters')
    H, jumps, inventory, resource, _ = build_model(p)
    Q = sp.kron(sp.eye(8), o['masks'][6], format='csr')
    currents = []; deformation = 0*H; hop_terms = []
    for e, ((i, j), theta) in enumerate(zip(PAIRS, (0., 0., -phi))):
        a = o['up'][i] @ o['up'][j].getH()
        old = p['J']*o['pl'][e]@(a+a.getH())
        h = p['J']*o['pl'][e]@(np.exp(1j*theta)*a+np.exp(-1j*theta)*a.getH())
        if e == 1:
            deformation = clean((v-1.)*Q@h); h += deformation
        H += h-old; hop_terms.append(clean(h))
        currents.append(clean(1j*(h@o['n'][j]-o['n'][j]@h)))
        if model == 'IC1_QS':
            jumps.append(('feed_'+EDGES[e], clean(np.sqrt(IC_ALPHA[e])*o['bp'][e]@o['c'][e])))
    H = clean(H+eps*o['n'][1]); N = np.array([(k//27).bit_count() for k in range(216)])
    space = Space(N)
    obs = {'J_h': currents[1]+currents[2], 'K': currents[1]-currents[2], 'F': .5*o['n'][2],
           'source': .5*(o['eye']-o['n'][0]), 'inventory': inventory}
    for m in range(8):
        for n in range(4):
            pmn = sp.diags(((N == n) & np.tile(o['mask_id'] == m, 8)).astype(float), format='csr')
            obs[f'p_m{m}_N{n}'] = pmn
            obs[f'h_m{m}_N{n}'] = clean(pmn@obs['J_h'])
            for tag, I in zip(('I01', 'I12', 'I02'), currents):
                obs[f'{tag}_m{m}_N{n}'] = clean(pmn@I)
    obs['P_m6_N2'] = obs['p_m6_N2']; obs['h6'] = obs['h_m6_N2']
    for name, (m, a, b) in TARGETS.items():
        # Tr(|b><a| rho) = rho[a,b], preserving the declared sign convention.
        obs['C_'+name] = sp.kron(ketbra(8, b, a), o['masks'][m], format='csr')
    adj = lambda a: clean(1j*(H@a-a@H)+sum(adjoint_jump(c, a) for _, c in jumps))
    checks = dict(H_hermiticity=maxabs(H-H.getH()), trace=maxabs(adj(o['eye'])),
                  node0=maxabs(adj(o['n'][0])+currents[0]+currents[2]-obs['source']),
                  node1=maxabs(adj(o['n'][1])-currents[0]+currents[1]),
                  node2=maxabs(adj(o['n'][2])-obs['J_h']+obs['F']),
                  resource=maxabs(adj(inventory)-resource),
                  deformation_support=maxabs((o['eye']-Q)@deformation),
                  deformation_commutator=maxabs(Q@deformation-deformation@Q))
    for m in range(8):
        pm = sp.kron(sp.eye(8), o['masks'][m], format='csr')
        obs[f'mask_balance_m{m}'] = adj(pm)
    if max(checks.values()) > 1e-11:
        raise ValueError('Full operator checks failed: '+str(checks))
    result = dict(H=H, jumps=jumps, space=space, observables=obs, currents=currents,
                  n=o['n'], case=case, checks=checks, deformation=deformation, hop_terms=hop_terms,
                  charge_checks=charge_checks(H, jumps, space))
    if assemble:
        result['L'] = liouvillian(H, jumps, space)
    return result

    rates = np.zeros((216, 216))
    masks = operators()['mask_id']
    for name in OPERATIONS[target]:
        m, a, b = TARGETS[name]
        ds = []
        for sign in (-1., 1.):
            x = np.zeros(8); x[a] = 1.; x[b] = sign
            ds.append(np.kron(x, (masks == m).astype(float)))
        rates += -.5*((ds[0][:, None]-ds[0][None, :])**2 - (ds[1][:, None]-ds[1][None, :])**2)
    return sp.diags(rates.ravel(order='F')[space.indices], format='csr')

"""Bordered sparse solves; state and response checks deliberately differ."""
import time
import numpy as np
from scipy import sparse as sp
from scipy.sparse.linalg import splu, spilu, LinearOperator, onenormest
from .compat import gmres
from e002_exp.model import maxabs

FLOOR = 1e-6


def refinement_check(primary, reference, values_primary, values_reference, limits):
    """Calibrated engineering acceptance thresholds; not a rigorous error bound."""
    state_difference = maxabs(np.asarray(primary)-np.asarray(reference))
    errors = {name: float(abs(value-values_reference[name])) for name, value in values_primary.items()
              if value is not None and values_reference.get(name) is not None}
    failed = [name for name, error in errors.items() if error > limits['observable_absolute']+
              limits['observable_relative']*max(abs(values_primary[name]), abs(values_reference[name]))]
    passed = state_difference <= limits['vector_inf'] and not failed
    return dict(pass_gate=bool(passed), vector_difference_inf=state_difference, observable_differences=errors,
                failed_observables=failed, limits=limits, status='PASS' if passed else 'NUMERICALLY_UNRESOLVED')


def residual(A, x, b):
    r = A@x-b
    return dict(absolute_inf=maxabs(r), relative_2=float(np.linalg.norm(r)/max(np.linalg.norm(b), 1.)))


class Bordered:
    def __init__(self, L, space, z=0j, method='lu', rtol=1e-11, atol=1e-13,
                 maxiter=400, restart=100, ilu_drop_tol=1e-5, ilu_fill_factor=10.):
        self.space = space; self.L = sp.csr_matrix(L); self.z = complex(z)
        self.A = self.L if z == 0 else (z*sp.eye(space.size, format='csr')-self.L)
        r = space.trace_row.T/space.d
        self.matrix = sp.bmat([[self.A, r], [space.trace_row, None]], format='csc')
        self.rtol, self.atol, self.maxiter, self.restart = rtol, atol, maxiter, restart
        self.method = method; self.factor = None; self.preconditioner = None
        self.iterations = 0; self.solves = 0; self.solve_seconds = 0.
        start = time.perf_counter()
        if method == 'lu':
            self.factor = splu(self.matrix)
        elif method == 'gmres_ilu':
            self.factor = spilu(self.matrix, drop_tol=ilu_drop_tol, fill_factor=ilu_fill_factor)
            self.preconditioner = LinearOperator(self.matrix.shape, matvec=self.factor.solve, dtype=complex)
        else:
            raise ValueError('Unsupported solver')
        self.factor_seconds = time.perf_counter()-start
        self.factor_nnz = int(self.factor.L.nnz+self.factor.U.nnz)
        self.factor_storage_bytes = int(sum(a.data.nbytes+a.indices.nbytes+a.indptr.nbytes for a in (self.factor.L, self.factor.U)))

    def solve(self, b=None, trace=0.):
        b = np.zeros(self.space.size, complex) if b is None else np.asarray(b, complex)
        if b.shape != (self.space.size,) or not np.isfinite(b).all():
            raise ValueError('Invalid RHS')
        if abs(self.space.trace(b)) > max(1e-11, 10*self.atol):
            raise ValueError('RHS must have zero trace')
        if self.z != 0 and trace != 0:
            raise ValueError('Nonzero frequency expects traceless response')
        rhs = np.r_[b, trace]; start = time.perf_counter(); iterations = 0
        if self.method == 'lu':
            x = self.factor.solve(rhs)
            # Residual correction, never a state projection or renormalization.
            for _ in range(2):
                correction = self.factor.solve(rhs-self.matrix@x)
                x += correction
        else:
            def count(_):
                nonlocal iterations
                iterations += 1
            x, info = gmres(self.matrix, rhs, M=self.preconditioner, rtol=self.rtol,
                            atol=self.atol, maxiter=self.maxiter, restart=self.restart,
                            callback=count, callback_type='pr_norm')
            if info != 0:
                raise ArithmeticError('GMRES did not converge: '+str(info))
        self.solve_seconds += time.perf_counter()-start; self.solves += 1; self.iterations += iterations
        checks = residual(self.A, x[:-1], b)
        checks.update(trace_error=float(abs(self.space.trace(x[:-1])-trace)), border_multiplier=float(abs(x[-1])), iterations=iterations)
        limit = max(10*self.atol, 10*self.rtol*max(1., np.linalg.norm(b)))
        if not np.isfinite(x).all() or max(checks['absolute_inf'], checks['trace_error'], checks['border_multiplier']) > limit:
            raise ArithmeticError('Original equation/trace residual failed: '+str(checks))
        return x[:-1], checks

    def condition_estimate(self):
        if self.method != 'lu':
            return dict(value=None, status='NOT_ESTIMATED_FOR_ITERATIVE_SOLVER')
        inverse = LinearOperator(self.matrix.shape, matvec=self.factor.solve,
                                 rmatvec=lambda x: self.factor.solve(x, trans='H'), dtype=complex)
        estimate = float(sp.linalg.norm(self.matrix, 1)*onenormest(inverse))
        return dict(value=estimate, status='ONE_NORM_ESTIMATE_NOT_RIGOROUS_ERROR_BOUND')

    def metrics(self):
        return dict(method=self.method, matrix_nnz=int(self.matrix.nnz), factor_nnz=self.factor_nnz,
                    factor_storage_bytes=self.factor_storage_bytes, factor_seconds=self.factor_seconds,
                    solves=self.solves, solve_seconds=self.solve_seconds, iterations=self.iterations)


def state_checks(L, space, x, tolerance=1e-10):
    rho = space.unpack(x)
    eigmin = min(float(np.linalg.eigvalsh((rho[np.ix_(g, g)]+rho[np.ix_(g, g)].conj().T)/2).min()) for g in space.groups)
    result = dict(trace_error=float(abs(space.trace(x)-1)), hermiticity=maxabs(rho-rho.conj().T),
                  minimum_eigenvalue=eigmin, Lrho_inf=maxabs(L@x), Lrho_2=float(np.linalg.norm(L@x)))
    result['pass'] = bool(result['trace_error'] <= tolerance and result['hermiticity'] <= tolerance
                          and eigmin >= -tolerance and result['Lrho_inf'] <= tolerance)
    return result


def values(model, vector):
    return {name: complex(model['space'].row(op)@vector) for name, op in model['observables'].items()}


def ratio(h, p, uh=None, up=None):
    if not np.isfinite([h, p]).all() or p <= FLOOR:
        return dict(value=None, uncertainty=None, status='SMALL_DENOMINATOR')
    j = h/p
    if uh is None or up is None:
        return dict(value=float(j), uncertainty=None, status='ERROR_UNRESOLVED')
    if min(uh, up) < 0 or not np.isfinite([uh, up]).all():
        raise ValueError('Invalid error estimates')
    if p-up <= FLOOR:
        return dict(value=None, uncertainty=None, status='DENOMINATOR_UNRESOLVED')
    return dict(value=float(j), uncertainty=float((uh+abs(j)*up)/(p-up)), status='PROPAGATED_ESTIMATE')


def contrast(left, center, right, delta):
    if delta <= 0:
        raise ValueError('delta must be positive')
    if any(x['value'] is None for x in (left, center, right)):
        return dict(D=None, d=None, curvature=None, uncertainty_D=None, status='UNDEFINED')
    D = center['value']-(left['value']+right['value'])/2
    u = None if any(x['uncertainty'] is None for x in (left, center, right)) else center['uncertainty']+(left['uncertainty']+right['uncertainty'])/2
    return dict(D=float(D), d=None if abs(center['value']) <= FLOOR else float(-D/center['value']),
                curvature=float(-2*D/delta**2), uncertainty_D=u,
                center_below_both=bool(center['value'] < min(left['value'], right['value'])),
                status='ERROR_UNRESOLVED' if u is None else 'PROPAGATED_ESTIMATE')

    cpuinfo = read_optional('/proc/cpuinfo') or ''

    mem = dict(x.split(':', 1) for x in (read_optional('/proc/meminfo') or '').splitlines() if ':' in x)

        from threadpoolctl import threadpool_info
        pools = [{k: p.get(k) for k in ('user_api', 'internal_api', 'num_threads', 'version', 'architecture')} for p in threadpool_info()]

def memory_preflight(budget, current):
    required = ('peak_additional_bytes', 'reserve_bytes', 'measured_peak_rss_bytes', 'simultaneous_factor_storage_bytes')
    if any(isinstance(budget.get(k), bool) or not isinstance(budget.get(k), (int, float)) or not np.isfinite(budget[k]) or budget[k] <= 0 for k in required):
        raise ValueError('Memory budget lacks measured peak/factor storage and positive reserve')
    if budget['peak_additional_bytes'] < budget['simultaneous_factor_storage_bytes']:
        raise ValueError('Memory budget cannot be smaller than simultaneous factors')
    if budget['peak_additional_bytes'] < budget['measured_peak_rss_bytes']:
        raise ValueError('Memory forecast below measured reference peak')
    try:
        available = int(current['memory_available'].split()[0])*1024
    except (KeyError, TypeError, ValueError):
        raise ValueError('Host available memory is not measurable') from None
    if current.get('memory_limit') is not None and current.get('memory_current') is not None:
        try:
            limit=int(current['memory_limit']); used=int(current['memory_current'])
        except (TypeError,ValueError):
            raise ValueError('Cgroup memory values are not finite limits') from None
        mode='CGROUP_AND_HOST'
    else:
        limits=current.get('address_space_rlimit',[])
        used=current.get('process_vm_size_bytes')
        peak_vm=budget.get('measured_peak_vm_bytes')
        if (current.get('cgroup_visibility') != 'NOT_MOUNTED_IN_JOB_NAMESPACE' or len(limits)!=2
                or any(type(x) is not int or x<=0 for x in limits) or limits[0]!=limits[1]
                or type(used) is not int or used<=0 or not isinstance(peak_vm,(int,float))
                or not np.isfinite(peak_vm) or peak_vm<=0 or budget['peak_additional_bytes']<peak_vm):
            raise ValueError('Missing finite enforced address-space limit and measured virtual-memory budget')
        limit=limits[0];mode='HARD_RLIMIT_AS_AND_HOST_CGROUP_UNOBSERVED'
    remaining = min(limit-used, available)
    needed = budget['peak_additional_bytes']+budget['reserve_bytes']
    if needed > remaining:
        raise MemoryError(f'Memory budget needs {needed} bytes, available {remaining}')
    return dict(needed_bytes=needed, available_bytes=remaining, status='PASS',resource_mode=mode,
                caveat='Unknown ancestor cgroup cannot be measured inside namespace' if mode.startswith('HARD') else None)

    for path in kernel_files():

def compare_environment(reference, current):
    required = ('python', 'platform', 'machine', 'cpu_model', 'logical_cpus', 'affinity',
                'numpy', 'scipy', 'threadpools', 'requested_threads', 'blas_library_identity',
                'cgroup_visibility', 'address_space_rlimit')
    missing = [k for k in required if reference.get(k) is None or current.get(k) is None]
    differences = [k for k in required if reference.get(k) != current.get(k)]
    for k in ('cpu_quota','memory_limit'):
        if reference.get(k)!=current.get(k):differences.append(k)
    for value in (reference,current):
        if value.get('cgroup_visibility')=='NOT_MOUNTED_IN_JOB_NAMESPACE':
            limits=value.get('address_space_rlimit',[])
            if len(limits)!=2 or any(type(x) is not int or x<=0 for x in limits) or limits[0]!=limits[1]:missing.append('enforced_address_space_limit')
        elif value.get('cpu_quota') is None or value.get('memory_limit') is None:
            missing.append('cgroup_limits')
        if not value.get('blas_library_identity'):missing.append('blas_library_identity')
    return dict(comparable=not missing and not differences, missing=missing, differences=differences,
                note='Matching manifest does not eliminate unobserved host/cgroup contention; inspect runtime deviations')


def runtime_gate(forecast, elapsed, expected_counts, actual_counts, environment_match, calibrated):
    complete = all(k in forecast and np.isfinite(forecast[k]) for k in ('expected_seconds', 'lower_seconds', 'upper_seconds'))
    valid = complete and 0 < forecast['expected_seconds'] and 0 <= forecast['lower_seconds'] <= forecast['expected_seconds'] <= forecast['upper_seconds']
    if not valid:
        raise ValueError('Invalid declared runtime forecast')
    if not calibrated or not environment_match:
        status = 'NOT_EVALUABLE'
    elif expected_counts != actual_counts or not forecast['lower_seconds'] <= elapsed <= forecast['upper_seconds']:
        status = 'REVIEW_REQUIRED'
    else:
        status = 'PASS'
    return dict(**forecast, actual_seconds=elapsed, actual_over_expected=elapsed/forecast['expected_seconds'],
                deviation_seconds=elapsed-forecast['expected_seconds'], status=status,
                expected_counts=expected_counts, actual_counts=actual_counts,
                environment_comparable=environment_match, calibrated=calibrated,
                scope='Worker internal measured interval; enclosing process duration must be checked separately')
